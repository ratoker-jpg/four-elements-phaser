     3→ * Phaser Graphics primitives following the camera projection contract.

   387→  // ─── Frame sync ──────────────────────────────────────────────────
   388→
   389→  /**
   390→   * Sync blockout vehicle rendering from the current blockout vehicle state.
   391→   * Called each frame. Destroys stale graphics, creates new ones for new vehicles.
   392→   */
   393→  syncFromState(vehicles: BlockoutVehicleState[]): void {
   394→    const activeIds = new Set<string>();
   395→
   396→    // TURRET-HULL-CONTRACT-PR-F2: prepare the dedicated high-depth anchor
   397→    // diagnostic layer once per frame. Cleared here, drawn per-vehicle in
   398→    // drawTurretAnchorDiagnostic (called from renderVehicle). Kept on its own
   399→    // graphics object so the markers are never occluded by the turret sprite.
   400→    if (this.turretAnchorDebug && this.isDevtoolsActive()) {
   401→      if (!this.anchorGraphics) {
   402→        this.anchorGraphics = this.scene.add.graphics();
   403→        this.anchorGraphics.setDepth(ANCHOR_DEBUG_DEPTH);
   404→      }
   405→      this.anchorGraphics.clear();
   406→    }
   407→
   408→    // TURRET-HULL-CONTRACT-PR-F2: prepare the dedicated high-depth socket
   409→    // calibration layer once per frame. Cleared here, drawn per-vehicle in
   410→    // drawTurretSocketCalibration (called from renderVehicle).
   411→    if (this.socketCalibrate && this.isDevtoolsActive()) {
   412→      if (!this.socketCalibGraphics) {
   413→        this.socketCalibGraphics = this.scene.add.graphics();
   414→        this.socketCalibGraphics.setDepth(ANCHOR_DEBUG_DEPTH);
   415→      }
   416→      this.socketCalibGraphics.clear();
   417→    }
   418→
   419→    for (const vehicle of vehicles) {
   420→      activeIds.add(vehicle.id);
   421→
   422→      let g = this.vehicleGraphics.get(vehicle.id);
   423→      if (!g) {
   424→        g = this.scene.add.graphics();
   425→        g.setDepth(BLOCKOUT_DEPTH);
   426→        this.vehicleGraphics.set(vehicle.id, g);
   427→      }
   428→
   429→      // Check for generated hull sprite
   430→      // PIM-HULL-WASP-DIR-MAP-01: When calibration is active and vehicle is Wasp,
   431→      // use the forced visual dir16 instead of the normal pipeline.
   432→      // Arena/devtools-only: forced direction is gated to devtools mode.
   433→      const isWaspCalibrating = this.isDevtoolsActive()
   434→        && isCalibrationActive()
   435→        && bodyIdToGeneratedHullId(vehicle.bodyId) === 'wasp'
   436→        && isOverrideActive();
   437→
   438→      let hullKey: string | null;
   439→      if (isWaspCalibrating) {
   440→        const forcedDir16 = getForcedVisualDir16() as GeneratedHullDir16Index;
   441→        hullKey = resolveGeneratedHullKeyForced(
   442→          this.scene, vehicle.bodyId, vehicle.faction,
   443→          vehicle.modificationLevel, forcedDir16,
   444→        );
   445→      } else {
   446→        hullKey = resolveGeneratedHullKey(
   447→          this.scene, vehicle.bodyId, vehicle.faction,
   448→          vehicle.modificationLevel, vehicle.bodyAngle,
   449→        );
   450→      }
   451→      const useGeneratedHull = hullKey !== null;
   452→
   453→      // Manage hull sprite lifecycle
   454→      let hullSprite = this.vehicleHullSprites.get(vehicle.id);
   455→      if (useGeneratedHull && hullKey) {
   456→        if (!hullSprite) {
   457→          hullSprite = this.scene.add.image(0, 0, hullKey);
   458→          hullSprite.setScale(GENERATED_HULL_SCALE);
   459→          hullSprite.setOrigin(GENERATED_HULL_ORIGIN_X, GENERATED_HULL_ORIGIN_Y);
   460→          hullSprite.setDepth(BLOCKOUT_DEPTH + HULL_SPRITE_DEPTH_BIAS);
   461→          this.vehicleHullSprites.set(vehicle.id, hullSprite);
   462→          if (!this.generatedHullLogged) {
   463→            console.log(`[BlockoutVehicleRenderer] Using generated hull sprite for ${vehicle.bodyId}+${vehicle.weaponId}`);
   464→            this.generatedHullLogged = true;
   465→          }
   466→        } else {
   467→          hullSprite.setTexture(hullKey);
   468→        }
   469→      } else {
   470→        if (hullSprite) {
   471→          hullSprite.destroy();
   472→          this.vehicleHullSprites.delete(vehicle.id);
   473→        }
   474→        if (!this.blockoutFallbackLogged) {
   475→          console.log(`[BlockoutVehicleRenderer] No generated hull for ${vehicle.bodyId} — blockout fallback`);
   476→          this.blockoutFallbackLogged = true;
   477→        }
   478→      }
   479→
   480→      // ── TURRET-HULL-CONTRACT-PR-F2 / FIXUP-5: Turret sprite lifecycle ──
   481→      // Resolve turret sprite key using generated 16-dir/512px resolver.
   482→      // Only attempt when a generated hull sprite is active (Wasp+Smoky path).
   483→      // FIXUP-5: Uses resolveGeneratedTurretKey (16-dir, 512×512) instead of
   484→      // the legacy resolveModularTurretSpriteKey (8-dir, 256×256).
   485→      let turretKey: string | null = null;
   486→      if (useGeneratedHull) {
   487→        const turretVisualDir16 = turretAngleToVisualDir16(vehicle.turretAngle, vehicle.weaponId);
   488→        turretKey = resolveGeneratedTurretKey(
   489→          this.scene, vehicle.weaponId, vehicle.faction, vehicle.modificationLevel,
   490→          turretVisualDir16,
   491→        );
   492→      }
   493→
   494→      // Resolve mounting data FIRST — this determines whether a real turret
   495→      // sprite can be shown. The adapter checks the full contract:
   496→      // texture key + directional pivot + socket profile + computed offset.
   497→      // If any piece is missing, useRealTurretSprite is false.
   498→      //
   499→      // FIXUP-5: Turret source dimensions are now 512×512 (generated),
   500→      // and turret scale is GENERATED_TURRET_SCALE (0.12) matching hull scale.
   501→      const mountingData = resolveTurretSpriteMountingData({
   502→        textureKey: turretKey,
   503→        weaponId: vehicle.weaponId,
   504→        bodyId: vehicle.bodyId,
   505→        modificationLevel: vehicle.modificationLevel,
   506→        turretAngle: vehicle.turretAngle,
   507→        bodyAngle: vehicle.bodyAngle,
   508→        sourceSizes: {
   509→          hullSourceWidthPx: 512,
   510→          hullSourceHeightPx: 512,
   511→          turretSourceWidthPx: GENERATED_TURRET_SOURCE_WIDTH,
   512→          turretSourceHeightPx: GENERATED_TURRET_SOURCE_HEIGHT,
   513→        },
   514→        scaleFactors: { hullScale: GENERATED_HULL_SCALE, turretScale: GENERATED_TURRET_SCALE },
   515→      });
   516→
   517→      // TURRET-HULL-CONTRACT-PR-F2: capture mounting data for the anchor
   518→      // diagnostic overlay (drawn later in renderVehicle).
   519→      this.lastMountingData.set(vehicle.id, mountingData);
   520→
   521→      let turretSprite = this.vehicleTurretSprites.get(vehicle.id);
   522→      if (mountingData.useRealTurretSprite && mountingData.textureKey !== null) {
   523→        // Full contract exists — create or update real turret sprite
   524→        const resolvedKey = mountingData.textureKey;
   525→        if (!turretSprite) {
   526→          turretSprite = this.scene.add.image(0, 0, resolvedKey);
   527→          turretSprite.setScale(GENERATED_TURRET_SCALE);
   528→          turretSprite.setOrigin(0.5, 0.5); // IMPORTANT: origin stays centered, never set to pivot
   529→          turretSprite.setDepth(BLOCKOUT_DEPTH + TURRET_SPRITE_DEPTH_BIAS);
   530→          this.vehicleTurretSprites.set(vehicle.id, turretSprite);
   531→          if (!this.turretSpriteLogged) {
   532→            console.log(`[BlockoutVehicleRenderer] Using real turret sprite for ${vehicle.weaponId}`);
   533→            this.turretSpriteLogged = true;
   534→          }
   535→        } else {
   536→          turretSprite.setTexture(resolvedKey);
   537→        }
   538→
   539→        // Position turret sprite relative to hull sprite using mounting offset
   540→        if (hullSprite && mountingData.offsetFromHullCenter) {
   541→          // Apply hull placement offset (same as hull sprite uses)
   542→          const hullId = bodyIdToGeneratedHullId(vehicle.bodyId);
   543→          const placementOffset = hullId ? getGeneratedHullPlacementOffset(hullId) : { offsetX: 0, offsetY: 0 };
   544→
   545→          // Hull sprite center position (same calculation as hull positioning)
   546→          const recoilBodyOffset = vehicle.recoilBodyOffset ?? 0;
   547→          const bodyAngle = vehicle.bodyAngle;
   548→          const bodyImpulseX = -Math.cos(bodyAngle) * recoilBodyOffset;
   549→          const bodyImpulseY = -Math.sin(bodyAngle) * recoilBodyOffset;
   550→          let hullCx = vehicle.worldX + this.offset.x + bodyImpulseX;
   551→          let hullCy = vehicle.worldY + this.offset.y + bodyImpulseY;
   552→
   553→          // Apply same placement offset as hull sprite
   554→          hullCx += placementOffset.offsetX;
   555→          hullCy += placementOffset.offsetY;
   556→
   557→          // Apply same Wasp debug placement offset if active
   558→          const isWaspPlacement = this.isDevtoolsActive()
   559→            && isWaspPlacementActive()
   560→            && bodyIdToGeneratedHullId(vehicle.bodyId) === 'wasp';
   561→          if (isWaspPlacement) {
   562→            hullCx += getWaspDebugOffsetX();
   563→            hullCy += getWaspDebugOffsetY();
   564→          }
   565→
   566→          // Position turret sprite at hull center + mounting offset
   567→          turretSprite.setPosition(
   568→            hullCx + mountingData.offsetFromHullCenter.x,
   569→            hullCy + mountingData.offsetFromHullCenter.y,
   570→          );
   571→        }
   572→      } else {
   573→        // No full contract — destroy real turret sprite if it exists, use procedural fallback
   574→        if (turretSprite) {
   575→          turretSprite.destroy();
   576→          this.vehicleTurretSprites.delete(vehicle.id);
   577→        }
   578→      }
   579→
   580→      // Determine selection/hover state for this vehicle
   581→      const isSelected = vehicle.id === this._selectedVehicleId;
   582→      const isHovered = vehicle.id === this._hoveredVehicleId;
   583→
   584→      // Redraw this vehicle
   585→      this.renderVehicle(g, vehicle, isSelected, isHovered);
   586→
   587→      // Debug label
   588→      let label = this.debugLabels.get(vehicle.id);
   589→      if (!label && this.showDebugLabels) {
   590→        label = this.scene.add.text(0, 0, '', {
   591→          fontSize: '9px',
   592→          color: DEBUG_LABEL_COLOR,
   593→          backgroundColor: '#00000088',
   594→          padding: { x: 2, y: 1 },
   595→        });
   596→        label.setDepth(BLOCKOUT_DEPTH + 1);
   597→        label.setOrigin(0.5, 1);
   598→        this.debugLabels.set(vehicle.id, label);
   599→      }
   600→
   601→      if (label) {
   602→        const bodyCenter = computeBodyWorldCenter(vehicle, this.offset);
   603→
   604→        const selectedMarker = isSelected ? ' [SEL]' : '';
   605→        const speedMarker = vehicle.speed > 1 ? ` v=${Math.round(vehicle.speed)}` : '';
   606→        const hpMarker = vehicle.isDestroyed ? ' [DEAD]' : ` hp=${vehicle.hp}/${vehicle.maxHp}`;
   607→        label.setText(`${vehicle.bodyId}+${vehicle.weaponId}${selectedMarker}${hpMarker}${speedMarker}`);
   608→
   609→        // Position label above top face (body center + Z offset)
   610→        const labelZ = vehicle.isDestroyed ? 0 : BLOCKOUT_VEHICLE_BODY_Z + BLOCKOUT_TURRET_Z_OFFSET + 0.1;
   611→        const tilePos = unprojectScreenToGround(bodyCenter.x, bodyCenter.y, this.offset);
   612→        const labelPos = projectWorldPoint(tilePos.x, tilePos.y, labelZ, this.offset);
   613→        label.setPosition(labelPos.x, labelPos.y);
   614→        label.setVisible(this.showDebugLabels);
   615→      }
   616→    }
   617→
   618→    // ── CORE-STEP-06H+: Depth sorting ─────────────────────────────
   619→    // Sort all vehicles by isometric depth for correct occlusion.
   620→    // Vehicles with lower screen Y (further from camera) are drawn first.
   621→    const depthSortables: DepthSortable[] = vehicles
   622→      .filter(v => !v.isDestroyed && activeIds.has(v.id))
   623→      .map(v => ({
   624→        id: v.id,
   625→        type: 'unit' as const,
   626→        tx: v.tx,
   627→        ty: v.ty,
   628→        offsetX: this.offset.x,
   629→        offsetY: this.offset.y,
   630→      }));
   631→    const depthOrder = new Map<string, number>();
   632→    if (depthSortables.length > 0) {
   633→      const sorted = sortByDepth(depthSortables);
   634→      for (let i = 0; i < sorted.length; i++) {
   635→        depthOrder.set(sorted[i].sortable.id, i);
   636→      }
   637→    }
   638→
   639→    // Apply depth to each vehicle's graphics object
   640→    for (const vehicle of vehicles) {
   641→      const g = this.vehicleGraphics.get(vehicle.id);
   642→      if (g) {
   643→        const orderIdx = depthOrder.get(vehicle.id);
   644→        if (orderIdx !== undefined) {
   645→          // Use BLOCKOUT_DEPTH as base + sorted order for isometric correctness
   646→          g.setDepth(BLOCKOUT_DEPTH + orderIdx);
   647→        }
   648→      }
   649→      // Also update debug label depth
   650→      const label = this.debugLabels.get(vehicle.id);
   651→      if (label) {
   652→        const orderIdx = depthOrder.get(vehicle.id);
   653→        if (orderIdx !== undefined) {
   654→          label.setDepth(BLOCKOUT_DEPTH + orderIdx + 1);
   655→        }
   656→      }
   657→      // Also update hull sprite depth
   658→      const hullSprite = this.vehicleHullSprites.get(vehicle.id);
   659→      if (hullSprite) {
   660→        const orderIdx = depthOrder.get(vehicle.id);
   661→        if (orderIdx !== undefined) {
   662→          hullSprite.setDepth(BLOCKOUT_DEPTH + orderIdx + HULL_SPRITE_DEPTH_BIAS);
   663→        }
   664→      }
   665→      // TURRET-HULL-CONTRACT-PR-F2: Also update turret sprite depth
   666→      const turretSpriteObj = this.vehicleTurretSprites.get(vehicle.id);
   667→      if (turretSpriteObj) {
   668→        const orderIdx = depthOrder.get(vehicle.id);
   669→        if (orderIdx !== undefined) {
   670→          turretSpriteObj.setDepth(BLOCKOUT_DEPTH + orderIdx + TURRET_SPRITE_DEPTH_BIAS);
   671→        }
   672→      }
   673→      // Also update direction debug label depth
   674→      const dirLabel = this.directionDebugLabels.get(vehicle.id);
   675→      if (dirLabel) {
   676→        const orderIdx = depthOrder.get(vehicle.id);
   677→        if (orderIdx !== undefined) {
   678→          dirLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 10);
   679→        }
   680→      }
   681→      // Also update calibration label depth
   682→      const calibLabel = this.calibrationLabels.get(vehicle.id);
   683→      if (calibLabel) {
   684→        const orderIdx = depthOrder.get(vehicle.id);
   685→        if (orderIdx !== undefined) {
   686→          calibLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 20);
   687→        }
   688→      }
   689→      // Also update placement label depth
   690→      const placeLabel = this.placementLabels.get(vehicle.id);
   691→      if (placeLabel) {
   692→        const orderIdx = depthOrder.get(vehicle.id);
   693→        if (orderIdx !== undefined) {
   694→          placeLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 25);
   695→        }
   696→      }
   697→    }
   698→
   699→    // Clean up stale vehicles
   700→    for (const [id, g] of this.vehicleGraphics) {
   701→      if (!activeIds.has(id)) {
   702→        g.destroy();
   703→        this.vehicleGraphics.delete(id);
   704→      }
   705→    }
   706→    for (const [id, label] of this.debugLabels) {
   707→      if (!activeIds.has(id)) {
   708→        label.destroy();
   709→        this.debugLabels.delete(id);
   710→      }
   711→    }
   712→    // Clean up stale hull sprites
   713→    for (const [id, sprite] of this.vehicleHullSprites) {
   714→      if (!activeIds.has(id)) {
   715→        sprite.destroy();
   716→        this.vehicleHullSprites.delete(id);
   717→      }
   718→    }
   719→    // TURRET-HULL-CONTRACT-PR-F2: Clean up stale turret sprites
   720→    for (const [id, sprite] of this.vehicleTurretSprites) {
   721→      if (!activeIds.has(id)) {
   722→        sprite.destroy();
   723→        this.vehicleTurretSprites.delete(id);
   724→      }
   725→    }
   726→    // TURRET-HULL-CONTRACT-PR-F2: Clean up stale mounting data + anchor overlay
   727→    for (const id of this.lastMountingData.keys()) {
   728→      if (!activeIds.has(id)) this.lastMountingData.delete(id);
   729→    }
   730→    for (const id of this.anchorLoggedVehicles) {
   731→      if (!activeIds.has(id)) this.anchorLoggedVehicles.delete(id);
   732→    }
   733→    for (const [id, label] of this.anchorHullLabels) {
   734→      if (!activeIds.has(id)) {
   735→        label.destroy();
   736→        this.anchorHullLabels.delete(id);
   737→      }
   738→    }
   739→    for (const [id, label] of this.anchorTurretLabels) {
   740→      if (!activeIds.has(id)) {
   741→        label.destroy();
   742→        this.anchorTurretLabels.delete(id);
   743→      }
   744→    }
   745→    for (const [id, label] of this.anchorDeltaLabels) {
   746→      if (!activeIds.has(id)) {
   747→        label.destroy();
   748→        this.anchorDeltaLabels.delete(id);
   749→      }
   750→    }
   751→    // Clean up stale direction debug labels
   752→    for (const [id, label] of this.directionDebugLabels) {
   753→      if (!activeIds.has(id)) {
   754→        label.destroy();
   755→        this.directionDebugLabels.delete(id);
   756→      }
   757→    }
   758→    // Clean up stale calibration labels
   759→    for (const [id, label] of this.calibrationLabels) {
   760→      if (!activeIds.has(id)) {
   761→        label.destroy();
   762→        this.calibrationLabels.delete(id);
   763→      }
   764→    }
   765→    // Clean up stale placement labels
   766→    for (const [id, label] of this.placementLabels) {
   767→      if (!activeIds.has(id)) {
   768→        label.destroy();
   769→        this.placementLabels.delete(id);
   770→      }
   771→    }
   772→
   773→    // ── PIM-HULL-WASP-ANCHOR-MAP-01 fixup v3: On-screen calibration button panel ──
   774→    // Show the panel when placement calibration is active for a Wasp, hide otherwise.
   775→    // The panel is a fixed-screen UI that doesn't move with vehicles.
   776→    const shouldShowPanel = this.isDevtoolsActive() && isWaspPlacementActive();
   777→    if (shouldShowPanel && !this.placementPanel) {
   778→      this.placementPanel = new WaspPlacementCalibrationPanel(this.scene);
   779→      this.placementPanel.show();
   780→    } else if (!shouldShowPanel && this.placementPanel) {
   781→      this.placementPanel.hide();
   782→      this.placementPanel = null;
   783→    }
   784→  }
   785→
   786→  // ─── Vehicle rendering ──────────────────────────────────────────
   787→
   788→  private renderVehicle(g: Phaser.GameObjects.Graphics, vehicle: BlockoutVehicleState, isSelected: boolean, isHovered: boolean): void {
   789→    g.clear();
   790→
   791→    // If using generated hull sprite, skip blockout body rendering but keep overlays
   792→    const hullSprite = this.vehicleHullSprites.get(vehicle.id);
   793→    const skipBlockoutBody = hullSprite !== undefined;
   794→
   795→    // TURRET-HULL-CONTRACT-PR-F2: If a real turret sprite is active,
   796→    // skip procedural turret box + barrel rendering. Keep aim line if selected.
   797→    const turretSprite = this.vehicleTurretSprites.get(vehicle.id);
   798→    const skipBlockoutTurret = turretSprite !== undefined;
   799→
   800→    // Position hull sprite at vehicle center
   801→    if (hullSprite) {
   802→      const recoilBodyOffset = vehicle.recoilBodyOffset ?? 0;
   803→      const bodyAngle = vehicle.bodyAngle;
   804→      const bodyImpulseX = -Math.cos(bodyAngle) * recoilBodyOffset;
   805→      const bodyImpulseY = -Math.sin(bodyAngle) * recoilBodyOffset;
   806→      let spriteCx = vehicle.worldX + this.offset.x + bodyImpulseX;
   807→      let spriteCy = vehicle.worldY + this.offset.y + bodyImpulseY;
   808→
   809→      // PIM-HULL-WASP-ANCHOR-MAP-01: Apply debug placement offset for Wasp
   810→      // Arena/devtools-only: this offset is purely visual, affects sprite position only.
   811→      // Selection ring, movement, pathfinding are all unchanged.
   812→      const isWaspPlacement = this.isDevtoolsActive()
   813→        && isWaspPlacementActive()
   814→        && bodyIdToGeneratedHullId(vehicle.bodyId) === 'wasp';
   815→      if (isWaspPlacement) {
   816→        spriteCx += getWaspDebugOffsetX();
   817→        spriteCy += getWaspDebugOffsetY();
   818→      }
   819→
   820→      // PIM-WASP-SCALE-PLACEMENT-01: Apply per-hull placement offset.
   821→      // This is a permanent runtime offset (not the debug calibration offset)
   822→      // that centers the hull sprite correctly within the tile footprint
   823→      // after the scale reduction from 0.24 to 0.12.
   824→      // Only applies to hulls with generated sprites.
   825→      const hullId = bodyIdToGeneratedHullId(vehicle.bodyId);
   826→      if (hullId) {
   827→        const placement = getGeneratedHullPlacementOffset(hullId);
   828→        spriteCx += placement.offsetX;
   829→        spriteCy += placement.offsetY;
   830→      }
   831→
   832→      hullSprite.setPosition(spriteCx, spriteCy);
   833→      hullSprite.setDepth(BLOCKOUT_DEPTH + HULL_SPRITE_DEPTH_BIAS); // will be updated by depth sorting below
   834→    }
   835→
   836→    // ── Shared projected geometry (single source of truth) ────────
   837→    const geom = computeProjectedBlockoutVehicleGeometry(vehicle, this.offset);
   838→    const { halfW, halfH, mountTileOffset, turretHalfW, turretHalfH,
   839→            effectiveBarrelLength, effectiveTurretAngle, barrelZ,
   840→            barrelTipScreen, barrelStartScreen } = geom;
   841→
   842→    const bodyAngle = vehicle.bodyAngle;
   843→
   844→    // BLOCKOUT-04H+: Use continuous worldX/worldY + offset for position
   845→    // BLOCKOUT-05H+: Include recoil body impulse offset (shifts body backward)
   846→    const recoilBodyOffset = vehicle.recoilBodyOffset ?? 0;
   847→    const bodyImpulseX = -Math.cos(bodyAngle) * recoilBodyOffset;
   848→    const bodyImpulseY = -Math.sin(bodyAngle) * recoilBodyOffset;
   849→    const cx = vehicle.worldX + this.offset.x + bodyImpulseX;
   850→    const cy = vehicle.worldY + this.offset.y + bodyImpulseY;
   851→
   852→    // Tile-space body center (with impulse for visual rendering)
   853→    const tilePos = unprojectScreenToGround(cx, cy, this.offset);
   854→    // Tile-space mount position (shared offset + impulse-shifted body center)
   855→    const mountWorldX = tilePos.x + mountTileOffset.dx;
   856→    const mountWorldY = tilePos.y + mountTileOffset.dy;
   857→
   858→    // Faction colors
   859→    const bodyColor = FACTION_BODY_COLORS[vehicle.faction] ?? FACTION_BODY_COLORS.cyan;
   860→    const turretColor = FACTION_TURRET_COLORS[vehicle.faction] ?? FACTION_TURRET_COLORS.cyan;
   861→
   862→    // Barrel width from weapon profile
   863→    const wpProfile = getWeaponProfile(vehicle.weaponId);
   864→    const barrelWidth = wpProfile ? wpProfile.blockoutBarrelWidth : 2;
   865→
   866→    // ── Movement target marker (selected vehicle only) ──────────
   867→    if (isSelected && vehicle.hasMoveTarget) {
   868→      const targetScreenX = vehicle.targetWorldX + this.offset.x;
   869→      const targetScreenY = vehicle.targetWorldY + this.offset.y;
   870→
   871→      // Thin line from vehicle to target
   872→      g.lineStyle(1, MOVE_LINE_COLOR, MOVE_LINE_ALPHA);
   873→      g.beginPath();
   874→      g.moveTo(cx, cy);
   875→      g.lineTo(targetScreenX, targetScreenY);
   876→      g.strokePath();
   877→
   878→      // Projected ground diamond at target
   879→      g.lineStyle(1.5, MOVE_TARGET_COLOR, MOVE_TARGET_ALPHA);
   880→      drawProjectedGroundDiamond(g, targetScreenX, targetScreenY, MOVE_TARGET_DIAMOND_HALF_SIZE, this.offset);
   881→
   882→      // Projected crosshair
   883→      drawProjectedCrosshair(g, targetScreenX, targetScreenY, MOVE_TARGET_CROSSHAIR_ARM, this.offset);
   884→    }
   885→
   886→    // ── Selection highlight ring (projected ground-plane) ────────
   887→    if (isSelected) {
   888→      const pulse = 0.5 + 0.5 * Math.sin((this.scene.time.now % 800) / 800 * Math.PI * 2);
   889→      const alpha = 0.6 + 0.4 * pulse;
   890→
   891→      g.lineStyle(SELECTION_RING_WIDTH, SELECTION_RING_COLOR, alpha);
   892→      drawProjectedGroundRing(g, cx, cy, SELECTION_RING_WORLD_RADIUS, this.offset, 24);
   893→
   894→      // BLOCKOUT-10H+: Direction arrow outside the ring for orientation clarity
   895→      // Use screen-space arrow from body center along body angle
   896→      // Approximate ring edge in screen space for arrow placement
   897→      const ringEdgeDist = SELECTION_RING_WORLD_RADIUS * 38; // approximate pixel distance
   898→      const arrowBaseX = cx + Math.cos(bodyAngle) * ringEdgeDist;
   899→      const arrowBaseY = cy + Math.sin(bodyAngle) * ringEdgeDist;
   900→      const arrowTipX = cx + Math.cos(bodyAngle) * (ringEdgeDist + DIRECTION_ARROW_LENGTH);
   901→      const arrowTipY = cy + Math.sin(bodyAngle) * (ringEdgeDist + DIRECTION_ARROW_LENGTH);
   902→
   903→      // Arrow shaft
   904→      g.lineStyle(2, SELECTION_RING_COLOR, alpha);
   905→      g.beginPath();
   906→      g.moveTo(arrowBaseX, arrowBaseY);
   907→      g.lineTo(arrowTipX, arrowTipY);
   908→      g.strokePath();
   909→
   910→      // Arrow head
   911→      const headAngle1 = bodyAngle + Math.PI * 0.8;
   912→      const headAngle2 = bodyAngle - Math.PI * 0.8;
   913→      g.beginPath();
   914→      g.moveTo(arrowTipX, arrowTipY);
   915→      g.lineTo(arrowTipX + Math.cos(headAngle1) * DIRECTION_ARROW_HEAD, arrowTipY + Math.sin(headAngle1) * DIRECTION_ARROW_HEAD);
   916→      g.strokePath();
   917→      g.beginPath();
   918→      g.moveTo(arrowTipX, arrowTipY);
   919→      g.lineTo(arrowTipX + Math.cos(headAngle2) * DIRECTION_ARROW_HEAD, arrowTipY + Math.sin(headAngle2) * DIRECTION_ARROW_HEAD);
   920→      g.strokePath();
   921→    }
   922→
   923→    // ── Hover marker (projected ground-plane) ────────────────────
   924→    if (isHovered && !isSelected) {
   925→      g.lineStyle(1.5, HOVER_RING_COLOR, HOVER_RING_ALPHA);
   926→      drawProjectedGroundRing(g, cx, cy, HOVER_RING_WORLD_RADIUS, this.offset, 20);
   927→    }
   928→
   929→    // ── ARENA-03H+: Target indicator (enemy being targeted by selected ally) ──
   930→    const isTargeted = vehicle.id === this._targetedVehicleId;
   931→    if (isTargeted && !isSelected) {
   932→      const pulse = 0.5 + 0.5 * Math.sin((this.scene.time.now % 600) / 600 * Math.PI * 2);
   933→      const alpha = 0.5 + 0.5 * pulse;
   934→      g.lineStyle(2, 0xff4444, alpha); // Red targeting ring
   935→      drawProjectedGroundRing(g, cx, cy, SELECTION_RING_WORLD_RADIUS, this.offset, 24);
   936→
   937→      // Small crosshair in center
   938→      const crossSize = 0.12;
   939→      g.lineStyle(1.5, 0xff4444, alpha);
   940→      drawProjectedCrosshair(g, cx, cy, crossSize, this.offset);
   941→    }
   942→
   943→    // ── CORE-STEP-07H+: Target-lock status indicator on attacker ──
   944→    // Show a small colored dot above the turret when this vehicle has an active target-lock
   945→    if (vehicle.targetVehicleId && !vehicle.isDestroyed) {
   946→      // Target-lock active: show yellow dot above turret
   947→      const lockIndicatorZ = BLOCKOUT_VEHICLE_BODY_Z + BLOCKOUT_TURRET_Z_OFFSET + 0.3;
   948→      const tilePosLocal = unprojectScreenToGround(cx, cy, this.offset);
   949→      const lockPos = projectWorldPoint(tilePosLocal.x, tilePosLocal.y, lockIndicatorZ, this.offset);
   950→      g.fillStyle(0xffcc00, 0.9); // Yellow target-lock indicator
   951→      g.fillCircle(lockPos.x, lockPos.y, 3);
   952→    }
   953→
   954→    // ── ARENA-03H+: Enemy team indicator (small red diamond above HP bar) ──
   955→    if (vehicle.team === 'enemy' && !vehicle.isDestroyed) {
   956→      const indicatorZ = BLOCKOUT_VEHICLE_BODY_Z + BLOCKOUT_TURRET_Z_OFFSET + 0.2;
   957→      const indicatorPos = projectWorldPoint(tilePos.x, tilePos.y, indicatorZ, this.offset);
   958→      const indicatorSize = 0.08;
   959→      g.lineStyle(1, 0xff4444, 0.7);
   960→      drawProjectedGroundDiamond(g, indicatorPos.x, indicatorPos.y, indicatorSize, this.offset);
   961→    }
   962→
   963→    // ── BLOCKOUT-07H+: Destroyed vehicle rendering ────────────────
   964→    if (vehicle.isDestroyed) {
   965→      // Dimmed flat body on ground (no height)
   966→      const cosA = Math.cos(bodyAngle);
   967→      const sinA = Math.sin(bodyAngle);
   968→      const localCorners = [
   969→        { lx: -halfW, ly: -halfH },
   970→        { lx: halfW, ly: -halfH },
   971→        { lx: halfW, ly: halfH },
   972→        { lx: -halfW, ly: halfH },
   973→      ];
   974→      const basePts = localCorners.map(c => {
   975→        const wx = tilePos.x + c.lx * cosA - c.ly * sinA;
   976→        const wy = tilePos.y + c.lx * sinA + c.ly * cosA;
   977→        return projectWorldPoint(wx, wy, 0, this.offset);
   978→      });
   979→
   980→      g.fillStyle(bodyColor, 0.3);
   981→      g.beginPath();
   982→      g.moveTo(basePts[0].x, basePts[0].y);
   983→      for (let i = 1; i < basePts.length; i++) {
   984→        g.lineTo(basePts[i].x, basePts[i].y);
   985→      }
   986→      g.closePath();
   987→      g.fillPath();
   988→
