    83→/**
    84→ * ARCH-15A: Scene data for loading a saved game.
    85→ * When MainMenuScene loads a save, it passes this to GameScene.init().
    86→ *
    87→ * Fix 1: Includes saveSlotId so GameScene can update the same slot
    88→ * on subsequent saves instead of creating duplicates.
    89→ */
    90→export interface LoadSceneData {
    91→  loadedGameState: GameState;
    92→  mapId?: string;
    93→  /** Slot ID of the loaded save, used to update the same slot on re-save. */
    94→  saveSlotId?: string;
    95→}

    97→/**
    98→ * Infer mapStyle from terrain data in a loaded save.
    99→ * VISUAL-05A-PR2 fix: When a saved game is loaded, the setupConfig
   100→ * is built from DEFAULT_SETUP which has mapStyle='sand'. If the saved
   101→ * terrain contains any 'industrial' tiles, we must render as industrial
   102→ * or the terrain will be invisible (TERRAIN_KEY_MAP.industrial is '').
   103→ */
   104→function inferMapStyleFromTerrain(terrain: TerrainType[][]): MapStyle {
   105→  return terrain.some(row => row.some(t => t === 'industrial')) ? 'industrial' : 'sand';
   106→}

   201→  /**
   202→   * Receive scene data from MainMenu/NewGameSetup or from a loaded save.
   203→   * Called by Phaser before create().
   204→   *
   205→   * ARCH-15A: Accepts either a GameSetupConfig (new game) or a

   207→   * the saved state is used directly instead of createInitialState.

   209→  init(data: GameSetupConfig | LoadSceneData): void {
   210→    if ('loadedGameState' in data && data.loadedGameState) {
   211→      // VISUAL-05A-PR2 fix: Infer mapStyle from loaded terrain so industrial
   212→      // saves render correctly instead of falling back to sand (DEFAULT_SETUP).
   213→      const inferredStyle = inferMapStyleFromTerrain(data.loadedGameState.mapData.terrain);
   214→      this.setupConfig = {
   215→        ...DEFAULT_SETUP,
   216→        faction: data.loadedGameState.playerFaction,
   217→        mapId: data.mapId ?? 'customMap1',
   218→        mapStyle: inferredStyle,
   219→        resourceStyle: resolveResourceStyleForMapStyle(inferredStyle),
   220→      };
   221→      this.loadedGameState = data.loadedGameState;
   222→      // Fix 1: Preserve loaded slot ID for re-save
   223→      this.currentSaveSlotId = data.saveSlotId ?? null;
   224→    } else {
   225→      this.setupConfig = { ...DEFAULT_SETUP, ...data as GameSetupConfig };
   226→      this.loadedGameState = null;
   227→      // Fix 1: New game starts with no save slot
   228→      this.currentSaveSlotId = null;
   229→    }
   230→  }

   232→  create(): void {
   233→    // ARCH-14C: Apply saved UI scale on game start
   234→    applyUiScale(loadUiSettings().uiScale);
   235→
   236→    // ARCH-11B+12A fixup: Compute devtools/arena flags BEFORE any rendering.
   237→    // MENU-02: Mode detection now supports both URL params and GameSetupConfig.gameMode.
   238→    // URL params take priority (for QA shortcuts), but if no URL params are set,
   239→    // the config-based mode from menu selection is used.
   240→    const urlDevtools = isDevtoolsEnabled();
   241→    const urlArena = urlDevtools && isArenaEnabled();
   242→    const configDebug = this.setupConfig.gameMode === 'debug';
   243→    const configArena = this.setupConfig.gameMode === 'arena';
   244→
   245→    this.devtoolsActive = urlDevtools || configDebug || configArena;
   246→    this.arenaMode = urlArena || configArena;
   247→
   248→    // ARENA-01H+: Create ArenaModeContext — controls which subsystems are active
   249→    this.arenaCtx = createArenaModeContext(this.arenaMode);
   250→
   251→    // Determine the game state source — loaded save takes priority,
   252→    // then arena (dev-only), then normal setup config.
   253→    if (this.loadedGameState) {
   254→      // PHASER4-LOAD-02: Strip modular-combat entities from old saves in standard mode.
   255→      // Older saves may contain modular-combat entities whose textures are not loaded
   256→      // when devtools is disabled. Devtools/arena mode preserves them (textures loaded).
   257→      this.gameState = stripModularCombatFromState(this.loadedGameState, {
   258→        includeModularCombat: this.devtoolsActive,
   259→      });
   260→      this.loadedGameState = null;
   261→      const stripped = this.gameState.entities.filter(e => e.kind === 'modular-combat').length === 0
   262→        && (this.gameState.extraModularCombat?.length ?? 0) === 0;
   263→      if (!stripped) {
   264→        console.log('[GameScene] Loaded saved game state (modular-combat preserved: devtools mode).');
   265→      } else {
   266→        console.log('[GameScene] Loaded saved game state.');
   267→      }
   268→    } else if (this.arenaMode) {
   269→      // ARENA-01H+: Arena mode — clean standalone, no HQ/harvesters/resources/economy
   270→      const arenaMapData = createArenaMapData();
   271→      this.gameState = createInitialState(arenaMapData, this.setupConfig.faction, 'QA Arena', { includeModularCombat: true, arenaMode: true });
   272→      console.log('[GameScene] Arena mode active. Map: QA Arena (20x20) — clean standalone');
   273→    } else {
   274→      const mapData = getMapDataFromConfig(this.setupConfig);
   275→      const mapNameOverride = getMapDisplayName(this.setupConfig);
   276→      this.gameState = createInitialState(mapData, this.setupConfig.faction, mapNameOverride, { includeModularCombat: this.devtoolsActive });
   277→    }
   278→
   279→    // CORE-STEP-06H+: Initialize tile reservation map for grid movement
   280→    this.reservationMap = new TileReservationMap(this.gameState.mapWidth);
   281→
   282→    // ARCH-11A: Log faction for smoke test verification
   283→    console.log(`[GameScene] Faction: ${this.gameState.playerFaction}`);
   284→
   285→    // Verify all required assets are loaded
   286→    this.verifyAssets();
   287→
   288→    // Render terrain from GameState.mapData.terrain
   289→    // VISUAL-05A-PR2: Pass mapStyle to TerrainRenderer for industrial/sand rendering
   290→    const mapStyle: MapStyle = this.setupConfig.mapStyle ?? 'sand';
   291→    this.terrainRenderer = new TerrainRenderer(
   292→      this,
   293→      this.gameState.mapData.terrain,
   294→      this.gameState.mapWidth,
   295→      this.gameState.mapHeight,
   296→      mapStyle,
   297→    );
   298→
   299→    // VISUAL-05A-PR3: Create industrial frame/background renderer when industrial
   300→    if (mapStyle === 'industrial') {
   301→      this.industrialFrameRenderer = new IndustrialFrameRenderer(
   302→        this,
   303→        this.gameState.mapWidth,
   304→        this.gameState.mapHeight,
   305→      );
   306→    }
   307→
   308→    // Get offset for entity placement
   309→    const offset = mapOriginOffset(this.gameState.mapWidth, this.gameState.mapHeight);
   310→    this._offset = offset;
   311→
   312→    // TERRAIN-01: Grid lines removed — they reinforced the chessboard
   313→    // pattern and are no longer needed with improved terrain clustering.
   314→    // If needed for debugging, re-enable drawGridLines() temporarily.
   315→
   316→    // Render entities — static first, then dynamic
   317→    // ARENA-01H+: Arena mode has no Normal Game entities (empty entities, harvesters, resourceNodes)
   318→    const resourceStyle: ResourceStyle = this.setupConfig.resourceStyle ?? 'legacy';
   319→    this.entityRenderer = new EntityRenderer(this, offset, resourceStyle);
   320→    this.entityRenderer.renderStaticEntities(this.gameState.entities);
   321→    if (!this.arenaCtx.arenaMode) {
   322→      this.entityRenderer.renderDynamicInit(
   323→        this.gameState.harvesters,
   324→        this.gameState.resourceNodes,
   325→      );
   326→    }
   327→
   328→    // ARCH-11A: Log harvester animation readiness for smoke test verification
   329→    console.log('[GameScene] Harvester animation ready.');
   330→
   331→    // ARCH-07A: Building status renderer (separator progress, factory queue, construction labels)
   332→    this.buildingStatusRenderer = new BuildingStatusRenderer(this, offset);
   333→
   334→    // ARCH-13A: Feedback renderer for command indicators and resource flow
   335→    this.feedbackRenderer = new FeedbackRenderer(this, offset);
   336→
   337→    // ARCH-13C-LITE: Motion dust renderer — render-only movement dust
   338→    this.motionFxRenderer = new UnitMotionFxRenderer(this, offset);
   339→
   340→    // Setup camera
   341→    // VISUAL-05A-PR3: Use extended bounds when industrial frame is present
   342→    this.cameraControls = new CameraControls(this);
   343→    const bounds = this.industrialFrameRenderer
   344→      ? this.industrialFrameRenderer.getExtendedBounds()
   345→      : this.terrainRenderer.getBounds();
   346→    this.cameraControls.setBounds(bounds);
   347→
   348→    // ARENA-01H+: Center camera on map center for Arena, HQ for Normal Game
   349→    if (this.arenaCtx.arenaMode) {
   350→      const centerTx = Math.floor(this.gameState.mapWidth / 2);
   351→      const centerTy = Math.floor(this.gameState.mapHeight / 2);
   352→      const centerScreen = tileToScreen(centerTx, centerTy);
   353→      this.hqWorldX = centerScreen.x + offset.x;
   354→      this.hqWorldY = centerScreen.y + offset.y;
   355→    } else {
   356→      const hq = this.gameState.mapData.hq;
   357→      const hqCenterTx = hq.tx + 1;
   358→      const hqCenterTy = hq.ty + 1;
   359→      const hqScreen = tileToScreen(hqCenterTx, hqCenterTy);
   360→      this.hqWorldX = hqScreen.x + offset.x;
   361→      this.hqWorldY = hqScreen.y + offset.y;
   362→    }
   363→    this.cameraControls.centerOn(this.hqWorldX, this.hqWorldY);
   364→    this.cameraControls.bindResetKey('R', this.hqWorldX, this.hqWorldY);
   365→
   366→    // ARENA-01H+: Arena mode uses ArenaMenu instead of PlaytestHud
   367→    if (this.arenaCtx.showPlaytestHud) {
   368→      this.playtestHud = new PlaytestHud();
   369→    }
   370→
   371→    // ARENA-01H+: Create ArenaMenu if in Arena mode
   372→    if (this.arenaCtx.showArenaMenu) {
   373→      this.arenaMenu = new ArenaMenu();
   374→      this.arenaMenu.create({
   375→        onResetArena: () => {
   376→          this.scene.restart({ faction: this.setupConfig.faction, mapId: ARENA_MAP_ID, gameMode: 'arena' });
   377→        },
   378→        onClearUnits: () => {
   379→          // ARENA-01H+: Clear all blockout vehicles
   380→          if (this.gameState.blockoutVehicles) {
   381→            this.gameState.blockoutVehicles.length = 0;
   382→            this.gameState.blockoutObstacles = [];
   383→          }
   384→        },
   385→        onToggleHelp: () => {
   386→          // ARENA-04H+: Toggle both ArenaMenu help overlay and HUD renderer help
   387→          this.arenaMenu?.toggleHelp();
   388→          this.blockoutSandboxHudRenderer?.toggleHelp();
   389→        },
   390→        // ARENA-02H+: Placement mode callbacks
   391→        onPlaceUnit: () => {
   392→          const composer = this.arenaMenu?.getUnitComposer();
   393→          if (!composer) return;
   394→          const selections = composer.getSelections();
   395→          if (!selections.body || !selections.weapon) return;
   396→          // Sync selections to placement state
   397→          this.arenaPlacementState.selectedBody = selections.body;
   398→          this.arenaPlacementState.selectedWeapon = selections.weapon;
   399→          this.arenaPlacementState.selectedTeam = selections.team;
   400→          this.arenaPlacementState.selectedAiMode = selections.aiMode; // ARENA-05H+
   401→          const entered = enterPlacementMode(this.arenaPlacementState);
   402→          if (!entered) {
   403→            this.arenaMenu?.showPlacementFeedback('Select body and weapon first', false);
   404→          }
   405→        },
   406→        onCancelPlacement: () => {
   407→          cancelPlacementMode(this.arenaPlacementState);
   408→          this.hidePlacementMarker();
   409→        },
   410→        getPlacementState: () => this.arenaPlacementState,
   411→        // ARENA-04H+: Roster callbacks — select, target, deselect, clear target
   412→        getSelectedVehicleId: () => this.blockoutVehicleInputController?.selectedVehicleId ?? null,
   413→        getTargetVehicleId: () => {
   414→          const selectedId = this.blockoutVehicleInputController?.selectedVehicleId;
   415→          if (!selectedId) return null;
   416→          const selected = this.gameState.blockoutVehicles?.find(v => v.id === selectedId);
   417→          return selected?.targetVehicleId ?? null;
   418→        },
   419→        onSelectVehicle: (vehicleId: string) => {
   420→          // ARENA-04H+: Select a vehicle from roster click.
   421→          // In Arena mode: only allies can be selected as controllable.
   422→          const vehicles = this.gameState.blockoutVehicles;
   423→          if (!vehicles) return;
   424→          const vehicle = vehicles.find(v => v.id === vehicleId);
   425→          if (!vehicle) return;
   426→          if (vehicle.team === 'enemy') return; // Enemies are not controllable
   427→          // Use the input controller's proper selection method
   428→          this.blockoutVehicleInputController?.setSelectedVehicleId(vehicleId);
   429→        },
   430→        onAssignTarget: (targetVehicleId: string) => {
   431→          // ARENA-04H+: Assign a target from roster enemy click.
   432→          const selectedId = this.blockoutVehicleInputController?.selectedVehicleId;
   433→          if (!selectedId) return;
   434→          const vehicles = this.gameState.blockoutVehicles;
   435→          if (!vehicles) return;
   436→          const selected = vehicles.find(v => v.id === selectedId);
   437→          if (!selected || selected.team !== 'ally') return;
   438→          // Validate target exists and is enemy
   439→          const target = vehicles.find(v => v.id === targetVehicleId);
   440→          if (!target || target.team !== 'enemy') return;
   441→          selected.targetVehicleId = targetVehicleId;
   442→        },
   443→        onDeselectVehicle: () => {
   444→          // ARENA-04H+: Deselect current vehicle and clear its target.
   445→          this.blockoutVehicleInputController?.setSelectedVehicleId(null);
   446→        },
   447→        onClearTarget: () => {
   448→          // ARENA-04H+: Clear target on selected vehicle.
   449→          // CORE-STEP-08H+ FIXUP Blocker 3: Uses clearTargetAndWeaponState
   450→          // to properly cancel wind-up (Railgun) and drum bursts (Hammer).
   451→          const selectedId = this.blockoutVehicleInputController?.selectedVehicleId;
   452→          if (!selectedId) return;
   453→          const vehicles = this.gameState.blockoutVehicles;
   454→          if (vehicles) {
   455→            const selected = vehicles.find(v => v.id === selectedId);
   456→            if (selected) {
   457→              clearTargetAndWeaponState(selected);
   458→            }
   459→          }

   984→  /**
   985→   * Handle pointer down in placement mode.
   986→   * LMB: attempt placement. RMB: cancel placement mode.
   987→   */
   988→  private handlePlacementPointerdown(pointer: Phaser.Input.Pointer): void {
   989→    if (!this.arenaMode) return;
   990→    if (this.arenaPlacementState.mode !== 'placing') return;
   991→
   992→    // RMB cancels placement
   993→    if (pointer.rightButtonDown()) {
   994→      cancelPlacementMode(this.arenaPlacementState);
   995→      this.hidePlacementMarker();
   996→      return;
   997→    }
   998→
   999→    // LMB: attempt placement at click position
  1000→    if (pointer.leftButtonDown()) {
  1001→      const worldPoint = this.cameras.main.getWorldPoint(pointer.x, pointer.y);
  1002→
  1003→      const result = convertClickToPlacementTile(
  1004→        worldPoint.x, worldPoint.y,
  1005→        this._offset,
  1006→        this.gameState.mapWidth,
  1007→        this.gameState.mapHeight,
  1008→        this.gameState.blockoutVehicles ?? [],
  1009→      );
  1010→
  1011→      if (!result.valid) {
  1012→        this.arenaMenu?.showPlacementFeedback(result.reason ?? 'Invalid placement', false);
  1013→        return;
  1014→      }
  1015→
  1016→      // Spawn the vehicle
  1017→      // ARENA-05H+ fixup: Pass selected AI mode for enemy units
  1018→      const spawnResult = arenaSpawnVehicle(
  1019→        this.gameState,
  1020→        this.arenaPlacementState.selectedBody!,
  1021→        this.arenaPlacementState.selectedWeapon!,
  1022→        this.arenaPlacementState.selectedTeam,
  1023→        result.tx,
  1024→        result.ty,
  1025→        this.arenaPlacementState.selectedTeam === 'enemy' ? this.arenaPlacementState.selectedAiMode : undefined,
  1026→      );
  1027→
  1028→      if (spawnResult.success) {
  1029→        this.arenaMenu?.showPlacementFeedback(spawnResult.message, true);
  1030→        // Stay in placement mode for rapid placement — user can Esc/RMB to cancel
  1031→      } else {
  1032→        this.arenaMenu?.showPlacementFeedback(spawnResult.message, false);
  1033→      }
  1034→    }
  1035→  }

  1078→  /**
  1079→   * Draw a placement marker on the ground plane at the given tile position.
  1080→   * ARENA-02H+: Uses projectGroundPoint from camera projection contract
  1081→   * to project a diamond shape onto the ground plane. NOT a screen-space circle.
  1082→   */
  1083→  private drawPlacementMarker(tx: number, ty: number, valid: boolean): void {
  1084→    if (!this.placementMarker) return;
  1085→
  1086→    this.placementMarker.clear();
  1087→    this.placementMarker.setVisible(true);
  1088→
  1089→    // EXPERIMENT-OPUS-B1B2-01 / B1: Diamond centered on (tx, ty) tile center,
  1090→    // aligned with the same baseline used by spawn/terrain/selection.
  1091→    const topCorner = projectGroundPoint(tx - 0.5, ty - 0.5, this._offset);
  1092→    const rightCorner = projectGroundPoint(tx + 0.5, ty - 0.5, this._offset);
  1093→    const bottomCorner = projectGroundPoint(tx + 0.5, ty + 0.5, this._offset);
  1094→    const leftCorner = projectGroundPoint(tx - 0.5, ty + 0.5, this._offset);
  1095→
  1096→    // Draw the projected diamond outline
  1097→    const color = valid ? 0x64c8ff : 0xff5050; // cyan for valid, red for invalid
  1098→    const alpha = 0.6;
  1099→
  1100→    this.placementMarker.lineStyle(2, color, alpha);
  1101→    this.placementMarker.beginPath();
  1102→    this.placementMarker.moveTo(topCorner.x, topCorner.y);
  1103→    this.placementMarker.lineTo(rightCorner.x, rightCorner.y);
  1104→    this.placementMarker.lineTo(bottomCorner.x, bottomCorner.y);
  1105→    this.placementMarker.lineTo(leftCorner.x, leftCorner.y);
  1106→    this.placementMarker.closePath();
  1107→    this.placementMarker.strokePath();
  1108→
  1109→    // Fill with semi-transparent color
  1110→    this.placementMarker.fillStyle(color, 0.12);
  1111→    this.placementMarker.fillPath();
  1112→
  1113→    // EXPERIMENT-OPUS-B1B2-01 / B1: Crosshair at tile center (tx, ty)
  1114→    const center = projectGroundPoint(tx, ty, this._offset);
  1115→    this.placementMarker.lineStyle(1, color, alpha * 0.7);
  1116→    this.placementMarker.beginPath();
  1117→    this.placementMarker.moveTo(center.x - 6, center.y);
  1118→    this.placementMarker.lineTo(center.x + 6, center.y);
  1119→    this.placementMarker.moveTo(center.x, center.y - 4);
  1120→    this.placementMarker.lineTo(center.x, center.y + 4);
  1121→    this.placementMarker.strokePath();
  1122→  }

  1124→  /**
  1125→   * Get the continuous fire aim target for a vehicle.
  1126→   *
  1127→   * ARENA-03H+ fixup: In Arena mode, returns null when there is no valid target.
  1128→   * This prevents continuous fire from falling back to turret-angle direction
  1129→   * or mouse position when the target is missing/destroyed.
  1130→   *
  1131→   * In non-Arena devtools mode, always returns mouse/fallback position (never null).
  1132→   *
  1133→   * @returns Aim target coordinates, or null if Arena mode has no valid target
  1134→   */
  1135→  private getContinuousFireAimTarget(vehicle: BlockoutVehicleState): { x: number; y: number } | null {
  1136→    // For non-Arena devtools mode, fall back to mouse position (never null)
  1137→    if (!this.arenaMode) {
  1138→      return {
  1139→        x: this.blockoutVehicleInputController?.mouseWorldX ?? vehicle.worldX + this._offset.x,
  1140→        y: this.blockoutVehicleInputController?.mouseWorldY ?? vehicle.worldY + this._offset.y,
  1141→      };
  1142→    }
  1143→
  1144→    // Arena mode: use target-lock
  1145→    if (vehicle.targetVehicleId) {
  1146→      const vehicles = this.gameState.blockoutVehicles;
  1147→      const target = vehicles?.find(v => v.id === vehicle.targetVehicleId);
  1148→      if (target && !target.isDestroyed) {
  1149→        const targetCenter = computeBodyWorldCenter(target, this._offset as IsoPoint);
  1150→        return { x: targetCenter.x, y: targetCenter.y };
  1151→      }
  1152→      // Target gone/destroyed — clear it and return null (stop fire)
  1153→      vehicle.targetVehicleId = null;
  1154→      return null;
  1155→    }
  1156→
  1157→    // Arena mode with no target: do not fire, do not fall back to turret angle or mouse
  1158→    return null;
  1159→  }

  1161→  /** Update Arena turret aim/rest for allies and enemies using one shared pass. */
  1162→  private updateArenaTurretAiming(delta: number): void {
  1163→    if (!this.arenaMode) return;
  1164→    const vehicles = this.gameState.blockoutVehicles;
  1165→    if (!vehicles) return;
  1166→
  1167→    for (const vehicle of vehicles) {
  1168→      if (vehicle.isDestroyed) continue;
  1169→
  1170→      let desiredAngle = vehicle.bodyAngle;
  1171→
  1172→      if (vehicle.targetVehicleId) {
  1173→        const target = vehicles.find(v => v.id === vehicle.targetVehicleId);
  1174→        if (target && !target.isDestroyed) {
  1175→          const turretMount = computeProjectedTurretMountScreen(vehicle, this._offset as IsoPoint);
  1176→          const targetCenter = computeBodyWorldCenter(target, this._offset as IsoPoint);
  1177→          desiredAngle = angleFromTo(turretMount.x, turretMount.y, targetCenter.x, targetCenter.y);
  1178→        } else {
  1179→          clearTargetAndWeaponState(vehicle);
  1180→          desiredAngle = vehicle.bodyAngle;
  1181→        }
  1182→      }
  1183→
  1184→      rotateTurretToward(vehicle, desiredAngle, delta);
  1185→    }
  1186→  }
