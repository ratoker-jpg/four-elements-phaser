    28→/**
    29→ * Generate pipeline output path planning for a hull.
    30→ */
    31→function planHullOutputPaths(hullName: string, mLevel: number, faction: string, directions: number) {
    32→  const paths = [];
    33→  for (let dir = 0; dir < directions; dir++) {
    34→    const key = `${hullName}_m${mLevel}_hull_${faction}_dir${dir}`;
    35→    const outputSubdir = `tankviewer/hulls/${hullName}/m${mLevel}`;
    36→    const filename = `${hullName}_m${mLevel}_hull_${faction}_dir${dir}.png`;
    37→    paths.push({ key, outputSubdir, filename });
    38→  }
    39→  return paths;
    40→}
    41→
    42→/**
    43→ * Generate pipeline output path planning for a turret.
    44→ */
    45→function planTurretOutputPaths(turretName: string, mLevel: number, faction: string, directions: number) {
    46→  const paths = [];
    47→  const gameName = TURRET_NAME_MAP[turretName] || turretName;
    48→  for (let dir = 0; dir < directions; dir++) {
    49→    const key = `${gameName}_m${mLevel}_turret_${faction}_dir${dir}`;
    50→    const outputSubdir = `tankviewer/turrets/${gameName}/m${mLevel}`;
    51→    const filename = `${gameName}_m${mLevel}_turret_${faction}_dir${dir}.png`;
    52→    paths.push({ key, outputSubdir, filename });
    53→  }
    54→  return paths;
    55→}

    57→/**
    58→ * Generate a full pipeline output path plan.
    59→ */
    60→function generatePathPlan(directions = 16, factions = ['cyan']) {
    61→  const plan: Record<string, unknown> = {
    62→    version: 2,
    63→    pipeline: 'tankviewer-blender-isometric',
    64→    directions,
    65→    factions,
    66→    nameMapping: TURRET_NAME_MAP,
    67→    pilotTarget: {
    68→      hull: 'wasp',
    69→      mLevel: 0,
    70→      turret: 'smoky',
    71→      faction: 'cyan',
    72→      directions,
    73→    },
    74→    hulls: {} as Record<string, Record<string, Record<string, ReturnType<typeof planHullOutputPaths>>>>,
    75→    turrets: {} as Record<string, Record<string, unknown>>,
    76→  };
    77→
    78→  for (const hull of HULLS) {
    79→    (plan.hulls as Record<string, Record<string, Record<string, ReturnType<typeof planHullOutputPaths>>>>)[hull] = {};
    80→    for (const m of M_LEVELS) {
    81→      (plan.hulls as Record<string, Record<string, Record<string, ReturnType<typeof planHullOutputPaths>>>>)[hull][`m${m}`] = {};
    82→      for (const faction of factions) {
    83→        (plan.hulls as Record<string, Record<string, Record<string, ReturnType<typeof planHullOutputPaths>>>>)[hull][`m${m}`][faction] =
    84→          planHullOutputPaths(hull, m, faction, directions);
    85→      }
    86→    }
    87→  }
    88→
    89→  for (const turret of TURRETS) {
    90→    const gameName = TURRET_NAME_MAP[turret] || turret;
    91→    (plan.turrets as Record<string, Record<string, unknown>>)[turret] = { gameName };
    92→    for (const m of M_LEVELS) {
    93→      ((plan.turrets as Record<string, Record<string, unknown>>)[turret] as Record<string, Record<string, ReturnType<typeof planTurretOutputPaths>>>)[`m${m}`] = {};
    94→      for (const faction of factions) {
    95→        ((plan.turrets as Record<string, Record<string, unknown>>)[turret] as Record<string, Record<string, ReturnType<typeof planTurretOutputPaths>>>)[`m${m}`][faction] =
    96→          planTurretOutputPaths(turret, m, faction, directions);
    97→      }
    98→    }
    99→  }
   100→
   101→  return plan;
   102→}

   400→// ─── Auto-fit centering transform tests (Option B: wrapper group) ────
   401→
   402→describe('TankViewer web exporter auto-fit centering transform', () => {
   403→  /**
   404→   * Simulates the Option B wrapper-group transform used in
   405→   * tools/tankviewer-web-exporter/index.html:
   406→   *   child.position = -center
   407→   *   wrapper.scale = (normalizeScale, normalizeScale * zScale, normalizeScale)
   408→   *   Final world vertex = normalizeScale * (V_original - center)
   409→   *
   410→   * These tests verify the math without Three.js at runtime.
   411→   */
   412→  function computeNormalizeScale(size: { x: number; y: number; z: number }, targetSize: number): number {
   413→    const maxDim = Math.max(size.x, size.y, size.z);
   414→    if (maxDim <= 0 || !isFinite(maxDim)) return 1;
   415→    return targetSize / maxDim;
   416→  }
   417→
   418→  /**
   419→   * Simulate the wrapper-group transform for an axis-aligned bbox.
   420→   * Returns the final bbox { min, max } and derived properties.
   421→   */
   422→  function simulateWrapperTransform(
   423→    bboxMin: [number, number, number],
   424→    bboxMax: [number, number, number],
   425→    targetSize: number,
   426→    zScale: number = 1.0,
   427→  ) {
   428→    // Original size
   429→    const sizeX = bboxMax[0] - bboxMin[0];
   430→    const sizeY = bboxMax[1] - bboxMin[1];
   431→    const sizeZ = bboxMax[2] - bboxMin[2];
   432→    const normalizeScale = computeNormalizeScale({ x: sizeX, y: sizeY, z: sizeZ }, targetSize);
   433→
   434→    // Center of original bbox
   435→    const cx = (bboxMin[0] + bboxMax[0]) / 2;
   436→    const cy = (bboxMin[1] + bboxMax[1]) / 2;
   437→    const cz = (bboxMin[2] + bboxMax[2]) / 2;
   438→
   439→    // Option B: final world vertex = wrapper.scale * (V - center)
   440→    // wrapper.scale = (normalizeScale, normalizeScale * zScale, normalizeScale)
   441→    // child.position = (-cx, -cy, -cz)
   442→    // For each original vertex V, world = (normalizeScale * (V.x - cx),
   443→    //                                       normalizeScale * zScale * (V.y - cy),
   444→    //                                       normalizeScale * (V.z - cz))
   445→    //
   446→    // Since bbox is axis-aligned, the extreme vertices are the 8 corners.
   447→    // For each axis, the min corner stays min and max corner stays max after
   448→    // the centering (V - center is symmetric), so:
   449→    //   finalMin.x = normalizeScale * (bboxMin.x - cx)
   450→    //   finalMax.x = normalizeScale * (bboxMax.x - cx)
   451→    //   etc.
   452→
   453→    const finalMinX = normalizeScale * (bboxMin[0] - cx);
   454→    const finalMaxX = normalizeScale * (bboxMax[0] - cx);
   455→    const finalMinY = normalizeScale * zScale * (bboxMin[1] - cy);
   456→    const finalMaxY = normalizeScale * zScale * (bboxMax[1] - cy);
   457→    const finalMinZ = normalizeScale * (bboxMin[2] - cz);
   458→    const finalMaxZ = normalizeScale * (bboxMax[2] - cz);
   459→
   460→    const finalCenterX = (finalMinX + finalMaxX) / 2;
   461→    const finalCenterY = (finalMinY + finalMaxY) / 2;
   462→    const finalCenterZ = (finalMinZ + finalMaxZ) / 2;
   463→
   464→    const finalSizeX = finalMaxX - finalMinX;
   465→    const finalSizeY = finalMaxY - finalMinY;
   466→    const finalSizeZ = finalMaxZ - finalMinZ;
   467→
   468→    return {
   469→      normalizeScale,
   470→      finalBbox: {
   471→        min: [finalMinX, finalMinY, finalMinZ] as [number, number, number],

   585→  it('REGRESSION: wrapper transform center is at origin, unlike old buggy transform', () => {
   586→    // The old (buggy) transform did: position = -center, then scale on same object
   587→    // In Three.js: worldVertex = scale * V + (-center)
   588→    //   = scale * V - center
   589→    // For Wasp center ≈ (1124, 611, 50):
   590→    //   worldCenter = scale * center - center = center * (scale - 1)
   591→    //   ≈ (1124 * -0.9935) ≈ -1117 — far from origin!
   592→
   593→    const bboxMin = [992.38, 379.92, -40.71] as [number, number, number];
   594→    const bboxMax = [1256.38, 841.92, 140.79] as [number, number, number];
   595→    const cx = (bboxMin[0] + bboxMax[0]) / 2; // ≈ 1124.38
   596→    const cy = (bboxMin[1] + bboxMax[1]) / 2; // ≈ 610.92
   597→    const cz = (bboxMin[2] + bboxMax[2]) / 2; // ≈ 50.04
   598→
   599→    const sizeX = bboxMax[0] - bboxMin[0]; // 264
   600→    const sizeY = bboxMax[1] - bboxMin[1]; // 462
   601→    const sizeZ = bboxMax[2] - bboxMin[2]; // 181.5
   602→    const maxDim = Math.max(sizeX, sizeY, sizeZ); // 462
   603→    const normalizeScale = 3.0 / maxDim;
   604→
   605→    // Old buggy center: center * (normalizeScale - 1)
   606→    const buggyCenterX = cx * (normalizeScale - 1);
   607→    const buggyCenterY = cy * (normalizeScale - 1);
   608→    const buggyCenterZ = cz * (normalizeScale - 1);
   609→
   610→    // The buggy center is FAR from origin
   611→    const buggyDistance = Math.sqrt(buggyCenterX ** 2 + buggyCenterY ** 2 + buggyCenterZ ** 2);
   612→    expect(buggyDistance).toBeGreaterThan(100); // Buggy center is hundreds of units away
   613→
   614→    // The correct (wrapper) center is at origin
   615→    const result = simulateWrapperTransform(bboxMin, bboxMax, 3.0);
   616→    const correctDistance = Math.sqrt(
   617→      result.finalCenter[0] ** 2 + result.finalCenter[1] ** 2 + result.finalCenter[2] ** 2,
   618→    );
   619→    expect(correctDistance).toBeCloseTo(0, 4);
   620→  });

   625→describe('TankViewer web exporter config.xml parsing', () => {
   626→  /**
   627→   * Pure helpers matching the parseConfigXml() and inference functions in
   628→   * tools/tankviewer-web-exporter/index.html.
   629→   * Duplicated here to avoid importing browser-only code into Node tests.
   630→   * We use a minimal XML parser (no DOMParser in Node) to test the logic.
   631→   */
   632→
   633→  /**
   634→   * Infer asset name from model file path.
   635→   * e.g. "hulls/Wasp_0123.3ds" → "Wasp"
   636→   */
   637→  function inferAssetName(filePath: string, _kind: string): string {
   638→    if (!filePath) return '';
   639→    const basename = filePath.split('/').pop()!.replace(/\.[^.]+$/, '');
   640→    const name = basename.replace(/_\d{1,4}$/, '');
   641→    return name.charAt(0).toUpperCase() + name.slice(1);
   642→  }
   643→
   644→  /**
   645→   * Infer M-level from details/lightmap filename suffix.
   646→   * e.g. "Wasp_0_details.png" → 0, "Wasp_2_lightmap.jpg" → 2
   647→   */
   648→  function inferMLevel(filePath: string): number {
   649→    if (!filePath) return 0;
   650→    const match = filePath.match(/_(\d)_(?:details|lightmap)/i);
   651→    if (match) return parseInt(match[1]);
   652→    const match2 = filePath.match(/_(\d)\.[^.]+$/);
   653→    if (match2 && parseInt(match2[1]) <= 3) return parseInt(match2[1]);
   654→    return 0;
   655→  }
   656→
   657→  /**
   658→   * Get expected filename from a path.
   659→   */
   660→  function getExpectedFilename(filePath: string): string {
   661→    if (!filePath) return '';
   662→    return filePath.split('/').pop()!;
   663→  }
   664→
   665→  /**
   666→   * Parse config.xml text using a simple regex-based extractor.
   667→   * This mirrors the DOMParser-based parseConfigXml in the browser,
   668→   * but works in Node without a DOM.
   669→   */
   670→  function parseConfigXmlSimple(xmlText: string) {
   671→    // Extract camera-radius
   672→    // Real TankViewer config.xml uses <config>; fallback to <root> for synthetic tests
   673→    const rootMatch = xmlText.match(/<(?:config|root)[^>]*camera-radius="([^"]*)"[^>]*>/);
   674→    const cameraRadius = rootMatch ? parseFloat(rootMatch[1]) || 750 : 750;
   675→
   676→    // Extract hull model entries
   677→    const hulls: Array<{
   678→      file: string;
   679→      details: string;
   680→      lightmap: string;
   681→      assetName: string;
   682→      mLevel: number;
   683→      kind: string;
   684→    }> = [];
   685→    const hullModelRegex = /<model\s+([^>]*)\/>/g;
   686→    const hullsSection = xmlText.match(/<hulls>([\s\S]*?)<\/hulls>/)?.[1] || '';
   687→    let match;
   688→    while ((match = hullModelRegex.exec(hullsSection)) !== null) {
   689→      const attrs = match[1];
   690→      const fileMatch = attrs.match(/file="([^"]*)"/);
   691→      const detailsMatch = attrs.match(/details="([^"]*)"/);
   692→      const lightmapMatch = attrs.match(/lightmap="([^"]*)"/);
   693→      const entry = {
   694→        file: fileMatch?.[1] || '',
   695→        details: detailsMatch?.[1] || '',
   696→        lightmap: lightmapMatch?.[1] || '',
   697→        assetName: '',
   698→        mLevel: 0,
   699→        kind: 'hull',
   700→      };
   701→      entry.assetName = inferAssetName(entry.file, 'hull');
   702→      entry.mLevel = inferMLevel(entry.details || entry.lightmap);
   703→      hulls.push(entry);
   704→    }
   705→
   706→    // Extract turret model entries
   707→    const turrets: Array<{
   708→      file: string;
   709→      details: string;
   710→      lightmap: string;
   711→      assetName: string;
   712→      mLevel: number;
   713→      kind: string;
   714→    }> = [];
   715→    const turretsSection = xmlText.match(/<turrets>([\s\S]*?)<\/turrets>/)?.[1] || '';
   716→    const turretModelRegex = /<model\s+([^>]*)\/>/g;
   717→    while ((match = turretModelRegex.exec(turretsSection)) !== null) {
   718→      const attrs = match[1];
   719→      const fileMatch = attrs.match(/file="([^"]*)"/);
   720→      const detailsMatch = attrs.match(/details="([^"]*)"/);
   721→      const lightmapMatch = attrs.match(/lightmap="([^"]*)"/);
   722→      const entry = {
   723→        file: fileMatch?.[1] || '',
   724→        details: detailsMatch?.[1] || '',
   725→        lightmap: lightmapMatch?.[1] || '',
   726→        assetName: '',
   727→        mLevel: 0,
   728→        kind: 'turret',
   729→      };
   730→      entry.assetName = inferAssetName(entry.file, 'turret');
   731→      entry.mLevel = inferMLevel(entry.details || entry.lightmap);
   732→      turrets.push(entry);
   733→    }
   734→
   735→    return { cameraRadius, hulls, turrets, colormaps: [] };
   736→  }

  1021→describe('TankViewer web exporter file validation', () => {
  1022→  function getExpectedFilename(filePath: string): string {
  1023→    if (!filePath) return '';
  1024→    return filePath.split('/').pop()!;
  1025→  }
  1026→
  1027→  it('matching filename passes validation', () => {
  1028→    const actual = 'Wasp_0123.3ds';
  1029→    const expected = getExpectedFilename('hulls/Wasp_0123.3ds');
  1030→    expect(actual).toBe(expected);
  1031→  });
  1032→
  1033→  it('mismatched filename fails validation', () => {
  1034→    const actual = 'Hornet_0123.3ds';
  1035→    const expected = getExpectedFilename('hulls/Wasp_0123.3ds');
  1036→    expect(actual).not.toBe(expected);
  1037→  });
  1038→
  1039→  it('different M-level texture fails validation', () => {
  1040→    const actual = 'Wasp_1_details.png';
  1041→    const expected = getExpectedFilename('hulls/Wasp_0_details.png');
  1042→    expect(actual).not.toBe(expected);
  1043→  });
  1044→
  1045→  it('correct M-level lightmap passes validation', () => {
  1046→    const actual = 'Wasp_0_lightmap.jpg';
  1047→    const expected = getExpectedFilename('hulls/Wasp_0_lightmap.jpg');
  1048→    expect(actual).toBe(expected);
  1049→  });
  1050→
  1051→  it('empty expected filename means no validation needed', () => {
  1052→    const expected = getExpectedFilename('');

  1054→    // When expected is empty, no validation warning is needed
  1055→  });
