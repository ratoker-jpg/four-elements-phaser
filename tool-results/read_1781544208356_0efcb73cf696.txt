    87→/**
    88→ * ARCH-15A: Scene data for loading a saved game.
    89→ * When MainMenuScene loads a save, it passes this to GameScene.init().
    90→ *
    91→ * Fix 1: Includes saveSlotId so GameScene can update the same slot

    94→export interface LoadSceneData {
    95→  loadedGameState: GameState;
    96→  mapId?: string;
    97→  /** Slot ID of the loaded save, used to update the same slot on re-save. */
    98→  saveSlotId?: string;
    99→}

   101→/**
   102→ * Infer mapStyle from terrain data in a loaded save.
   103→ * VISUAL-05A-PR2 fix: When a saved game is loaded, the setupConfig
   104→ * is built from DEFAULT_SETUP which has mapStyle='sand'. If the saved
   105→ * terrain contains any 'industrial' tiles, we must render as industrial
   106→ * or the terrain will be invisible (TERRAIN_KEY_MAP.industrial is '').
   107→ */
   108→function inferMapStyleFromTerrain(terrain: TerrainType[][]): MapStyle {
   109→  return terrain.some(row => row.some(t => t === 'industrial')) ? 'industrial' : 'sand';
   110→}

   215→  /**
   216→   * Receive scene data from MainMenu/NewGameSetup or from a loaded save.
   217→   * Called by Phaser before create().
   218→   *
   219→   * ARCH-15A: Accepts either a GameSetupConfig (new game) or a
   220→   * LoadSceneData (loaded save). If loadedGameState is present,
   221→   * the saved state is used directly instead of createInitialState.

   223→  init(data: GameSetupConfig | LoadSceneData): void {
   224→    if ('loadedGameState' in data && data.loadedGameState) {
   225→      // VISUAL-05A-PR2 fix: Infer mapStyle from loaded terrain so industrial
   226→      // saves render correctly instead of falling back to sand (DEFAULT_SETUP).
   227→      const inferredStyle = inferMapStyleFromTerrain(data.loadedGameState.mapData.terrain);
   228→      this.setupConfig = {
   229→        ...DEFAULT_SETUP,
   230→        faction: data.loadedGameState.playerFaction,
   231→        mapId: data.mapId ?? 'customMap1',
   232→        mapStyle: inferredStyle,
   233→        resourceStyle: resolveResourceStyleForMapStyle(inferredStyle),
   234→      };
   235→      this.loadedGameState = data.loadedGameState;
   236→      // Fix 1: Preserve loaded slot ID for re-save
   237→      this.currentSaveSlotId = data.saveSlotId ?? null;
   238→    } else {
   239→      this.setupConfig = { ...DEFAULT_SETUP, ...data as GameSetupConfig };
   240→      this.loadedGameState = null;
   241→      // Fix 1: New game starts with no save slot
   242→      this.currentSaveSlotId = null;
   243→    }
   244→  }

   246→  create(): void {
   247→    // ARCH-14C: Apply saved UI scale on game start
   248→    applyUiScale(loadUiSettings().uiScale);
   249→
   250→    // ARCH-11B+12A fixup: Compute devtools/arena flags BEFORE any rendering.
   251→    // MENU-02: Mode detection now supports both URL params and GameSetupConfig.gameMode.
   252→    // URL params take priority (for QA shortcuts), but if no URL params are set,
   253→    // the config-based mode from menu selection is used.
   254→    const urlDevtools = isDevtoolsEnabled();
   255→    const urlArena = urlDevtools && isArenaEnabled();
   256→    const configDebug = this.setupConfig.gameMode === 'debug';
   257→    const configArena = this.setupConfig.gameMode === 'arena';
   258→
   259→    this.devtoolsActive = urlDevtools || configDebug || configArena;
   260→    this.arenaMode = urlArena || configArena;
   261→
   262→    // ARENA-01H+: Create ArenaModeContext — controls which subsystems are active
   263→    this.arenaCtx = createArenaModeContext(this.arenaMode);
   264→
   265→    // Determine the game state source — loaded save takes priority,
   266→    // then arena (dev-only), then normal setup config.
   267→    if (this.loadedGameState) {
   268→      // PHASER4-LOAD-02: Strip modular-combat entities from old saves in standard mode.
   269→      // Older saves may contain modular-combat entities whose textures are not loaded
   270→      // when devtools is disabled. Devtools/arena mode preserves them (textures loaded).
   271→      this.gameState = stripModularCombatFromState(this.loadedGameState, {
   272→        includeModularCombat: this.devtoolsActive,
   273→      });
   274→      this.loadedGameState = null;
   275→      const stripped = this.gameState.entities.filter(e => e.kind === 'modular-combat').length === 0
   276→        && (this.gameState.extraModularCombat?.length ?? 0) === 0;
   277→      if (!stripped) {
   278→        console.log('[GameScene] Loaded saved game state (modular-combat preserved: devtools mode).');
   279→      } else {
   280→        console.log('[GameScene] Loaded saved game state.');
   281→      }
   282→    } else if (this.arenaMode) {
   283→      // ARENA-01H+: Arena mode — clean standalone, no HQ/harvesters/resources/economy
   284→      const arenaMapData = createArenaMapData();
   285→      this.gameState = createInitialState(arenaMapData, this.setupConfig.faction, 'QA Arena', { includeModularCombat: true, arenaMode: true });
   286→      console.log('[GameScene] Arena mode active. Map: QA Arena (20x20) — clean standalone');
   287→    } else {
   288→      const mapData = getMapDataFromConfig(this.setupConfig);
   289→      const mapNameOverride = getMapDisplayName(this.setupConfig);
   290→      this.gameState = createInitialState(mapData, this.setupConfig.faction, mapNameOverride, { includeModularCombat: this.devtoolsActive });
   291→    }
   292→
   293→    // CORE-STEP-06H+: Initialize tile reservation map for grid movement
   294→    this.reservationMap = new TileReservationMap(this.gameState.mapWidth);
   295→
   296→    // ARCH-11A: Log faction for smoke test verification
   297→    console.log(`[GameScene] Faction: ${this.gameState.playerFaction}`);
   298→
   299→    // Verify all required assets are loaded
   300→    this.verifyAssets();
   301→
   302→    // Render terrain from GameState.mapData.terrain
   303→    // VISUAL-05A-PR2: Pass mapStyle to TerrainRenderer for industrial/sand rendering
   304→    const mapStyle: MapStyle = this.setupConfig.mapStyle ?? 'sand';
   305→    this.terrainRenderer = new TerrainRenderer(
   306→      this,
   307→      this.gameState.mapData.terrain,
   308→      this.gameState.mapWidth,
   309→      this.gameState.mapHeight,
   310→      mapStyle,
   311→    );
   312→
   313→    // VISUAL-05A-PR3: Create industrial frame/background renderer when industrial
   314→    if (mapStyle === 'industrial') {
   315→      this.industrialFrameRenderer = new IndustrialFrameRenderer(
   316→        this,
   317→        this.gameState.mapWidth,
   318→        this.gameState.mapHeight,
   319→      );
   320→    }
   321→
   322→    // Get offset for entity placement
   323→    const offset = mapOriginOffset(this.gameState.mapWidth, this.gameState.mapHeight);
   324→    this._offset = offset;
   325→
   326→    // TERRAIN-01: Grid lines removed — they reinforced the chessboard
   327→    // pattern and are no longer needed with improved terrain clustering.
   328→    // If needed for debugging, re-enable drawGridLines() temporarily.
   329→
   330→    // Render entities — static first, then dynamic
   331→    // ARENA-01H+: Arena mode has no Normal Game entities (empty entities, harvesters, resourceNodes)
   332→    const resourceStyle: ResourceStyle = this.setupConfig.resourceStyle ?? 'legacy';
   333→    this.entityRenderer = new EntityRenderer(this, offset, resourceStyle);
   334→    this.entityRenderer.renderStaticEntities(this.gameState.entities);
   335→    if (!this.arenaCtx.arenaMode) {
   336→      this.entityRenderer.renderDynamicInit(
   337→        this.gameState.harvesters,
   338→        this.gameState.resourceNodes,
   339→      );
   340→    }
   341→
   342→    // ARCH-11A: Log harvester animation readiness for smoke test verification
   343→    console.log('[GameScene] Harvester animation ready.');
   344→
   345→    // ARCH-07A: Building status renderer (separator progress, factory queue, construction labels)
   346→    this.buildingStatusRenderer = new BuildingStatusRenderer(this, offset);
   347→
   348→    // ARCH-13A: Feedback renderer for command indicators and resource flow
   349→    this.feedbackRenderer = new FeedbackRenderer(this, offset);
   350→
   351→    // ARCH-13C-LITE: Motion dust renderer — render-only movement dust
   352→    this.motionFxRenderer = new UnitMotionFxRenderer(this, offset);
   353→
   354→    // Setup camera
   355→    // VISUAL-05A-PR3: Use extended bounds when industrial frame is present
   356→    this.cameraControls = new CameraControls(this);
   357→    const bounds = this.industrialFrameRenderer
   358→      ? this.industrialFrameRenderer.getExtendedBounds()
   359→      : this.terrainRenderer.getBounds();
   360→    this.cameraControls.setBounds(bounds);
   361→
   362→    // ARENA-01H+: Center camera on map center for Arena, HQ for Normal Game
   363→    if (this.arenaCtx.arenaMode) {
   364→      const centerTx = Math.floor(this.gameState.mapWidth / 2);
   365→      const centerTy = Math.floor(this.gameState.mapHeight / 2);
   366→      const centerScreen = tileToScreen(centerTx, centerTy);
   367→      this.hqWorldX = centerScreen.x + offset.x;
   368→      this.hqWorldY = centerScreen.y + offset.y;
   369→    } else {
   370→      const hq = this.gameState.mapData.hq;
   371→      const hqCenterTx = hq.tx + 1;
   372→      const hqCenterTy = hq.ty + 1;
   373→      const hqScreen = tileToScreen(hqCenterTx, hqCenterTy);
   374→      this.hqWorldX = hqScreen.x + offset.x;
   375→      this.hqWorldY = hqScreen.y + offset.y;
   376→    }
   377→    this.cameraControls.centerOn(this.hqWorldX, this.hqWorldY);
   378→    this.cameraControls.bindResetKey('R', this.hqWorldX, this.hqWorldY);
   379→
   380→    // ARENA-01H+: Arena mode uses ArenaMenu instead of PlaytestHud
   381→    if (this.arenaCtx.showPlaytestHud) {
   382→      this.playtestHud = new PlaytestHud();
   383→    }
   384→
   385→    // ARENA-01H+: Create ArenaMenu if in Arena mode
   386→    if (this.arenaCtx.showArenaMenu) {
   387→      this.arenaMenu = new ArenaMenu();
   388→      this.arenaMenu.create({
   389→        onResetArena: () => {
   390→          this.scene.restart({ faction: this.setupConfig.faction, mapId: ARENA_MAP_ID, gameMode: 'arena' });
   391→        },
   392→        onClearUnits: () => {
   393→          // ARENA-01H+: Clear all blockout vehicles
   394→          if (this.gameState.blockoutVehicles) {
   395→            this.gameState.blockoutVehicles.length = 0;
   396→            this.gameState.blockoutObstacles = [];
   397→          }
   398→        },
   399→        onToggleHelp: () => {
   400→          // ARENA-04H+: Toggle both ArenaMenu help overlay and HUD renderer help
   401→          this.arenaMenu?.toggleHelp();
   402→          this.blockoutSandboxHudRenderer?.toggleHelp();
   403→        },
   404→        // ARENA-02H+: Placement mode callbacks
   405→        onPlaceUnit: () => {
   406→          const composer = this.arenaMenu?.getUnitComposer();
   407→          if (!composer) return;
   408→          const selections = composer.getSelections();
   409→          if (!selections.body || !selections.weapon) return;
   410→          // Sync selections to placement state
   411→          this.arenaPlacementState.selectedBody = selections.body;
   412→          this.arenaPlacementState.selectedWeapon = selections.weapon;
   413→          this.arenaPlacementState.selectedTeam = selections.team;
   414→          this.arenaPlacementState.selectedAiMode = selections.aiMode; // ARENA-05H+
   415→          const entered = enterPlacementMode(this.arenaPlacementState);
   416→          if (!entered) {
   417→            this.arenaMenu?.showPlacementFeedback('Select body and weapon first', false);
   418→          }
   419→        },
   420→        onCancelPlacement: () => {
   421→          cancelPlacementMode(this.arenaPlacementState);
   422→          this.hidePlacementMarker();
   423→        },
   424→        getPlacementState: () => this.arenaPlacementState,
   425→        // ARENA-04H+: Roster callbacks — select, target, deselect, clear target
   426→        getSelectedVehicleId: () => this.blockoutVehicleInputController?.selectedVehicleId ?? null,
   427→        getTargetVehicleId: () => {
   428→          const selectedId = this.blockoutVehicleInputController?.selectedVehicleId;
   429→          if (!selectedId) return null;
   430→          const selected = this.gameState.blockoutVehicles?.find(v => v.id === selectedId);
   431→          return selected?.targetVehicleId ?? null;
   432→        },
   433→        onSelectVehicle: (vehicleId: string) => {
   434→          // ARENA-04H+: Select a vehicle from roster click.
   435→          // In Arena mode: only allies can be selected as controllable.
   436→          const vehicles = this.gameState.blockoutVehicles;
   437→          if (!vehicles) return;
   438→          const vehicle = vehicles.find(v => v.id === vehicleId);
   439→          if (!vehicle) return;
   440→          if (vehicle.team === 'enemy') return; // Enemies are not controllable
   441→          // Use the input controller's proper selection method
   442→          this.blockoutVehicleInputController?.setSelectedVehicleId(vehicleId);
   443→        },
   444→        onAssignTarget: (targetVehicleId: string) => {
   445→          // ARENA-04H+: Assign a target from roster enemy click.
   446→          const selectedId = this.blockoutVehicleInputController?.selectedVehicleId;
   447→          if (!selectedId) return;
   448→          const vehicles = this.gameState.blockoutVehicles;
   449→          if (!vehicles) return;
   450→          const selected = vehicles.find(v => v.id === selectedId);
   451→          if (!selected || selected.team !== 'ally') return;
   452→          // Validate target exists and is enemy
   453→          const target = vehicles.find(v => v.id === targetVehicleId);
   454→          if (!target || target.team !== 'enemy') return;
   455→          selected.targetVehicleId = targetVehicleId;

  1041→   * Handle pointer down in placement mode.
  1042→   * LMB: attempt placement. RMB: cancel placement mode.
  1043→   */
  1044→  private handlePlacementPointerdown(pointer: Phaser.Input.Pointer): void {
  1045→    if (!this.arenaMode) return;
  1046→    if (this.arenaPlacementState.mode !== 'placing') return;
  1047→
  1048→    // RMB cancels placement
  1049→    if (pointer.rightButtonDown()) {
  1050→      cancelPlacementMode(this.arenaPlacementState);
  1051→      this.hidePlacementMarker();
  1052→      return;
  1053→    }
  1054→
  1055→    // LMB: attempt placement at click position
  1056→    if (pointer.leftButtonDown()) {
  1057→      const worldPoint = this.cameras.main.getWorldPoint(pointer.x, pointer.y);
  1058→
  1059→      const result = convertClickToPlacementTile(
  1060→        worldPoint.x, worldPoint.y,
  1061→        this._offset,
  1062→        this.gameState.mapWidth,
  1063→        this.gameState.mapHeight,
  1064→        this.gameState.blockoutVehicles ?? [],
  1065→      );
  1066→
  1067→      if (!result.valid) {
  1068→        this.arenaMenu?.showPlacementFeedback(result.reason ?? 'Invalid placement', false);
  1069→        return;
  1070→      }
  1071→
  1072→      // Spawn the vehicle
  1073→      // ARENA-05H+ fixup: Pass selected AI mode for enemy units
  1074→      const spawnResult = arenaSpawnVehicle(
  1075→        this.gameState,
  1076→        this.arenaPlacementState.selectedBody!,
  1077→        this.arenaPlacementState.selectedWeapon!,
  1078→        this.arenaPlacementState.selectedTeam,
  1079→        result.tx,
  1080→        result.ty,
  1081→        this.arenaPlacementState.selectedTeam === 'enemy' ? this.arenaPlacementState.selectedAiMode : undefined,
  1082→      );
  1083→
  1084→      if (spawnResult.success) {
  1085→        this.arenaMenu?.showPlacementFeedback(spawnResult.message, true);
  1086→        // Stay in placement mode for rapid placement — user can Esc/RMB to cancel
  1087→      } else {
  1088→        this.arenaMenu?.showPlacementFeedback(spawnResult.message, false);
  1089→      }
  1090→    }
  1091→  }

  1135→   * Draw a placement marker on the ground plane at the given tile position.
  1136→   * ARENA-02H+: Uses projectGroundPoint from camera projection contract
  1137→   * to project a diamond shape onto the ground plane. NOT a screen-space circle.
  1138→   */
  1139→  private drawPlacementMarker(tx: number, ty: number, valid: boolean): void {
  1140→    if (!this.placementMarker) return;
  1141→
  1142→    this.placementMarker.clear();
  1143→    this.placementMarker.setVisible(true);
  1144→
  1145→    // EXPERIMENT-OPUS-B1B2-01 / B1: Diamond centered on (tx, ty) tile center,
  1146→    // aligned with the same baseline used by spawn/terrain/selection.
  1147→    const topCorner = projectGroundPoint(tx - 0.5, ty - 0.5, this._offset);
  1148→    const rightCorner = projectGroundPoint(tx + 0.5, ty - 0.5, this._offset);
  1149→    const bottomCorner = projectGroundPoint(tx + 0.5, ty + 0.5, this._offset);
  1150→    const leftCorner = projectGroundPoint(tx - 0.5, ty + 0.5, this._offset);
  1151→
  1152→    // Draw the projected diamond outline
  1153→    const color = valid ? 0x64c8ff : 0xff5050; // cyan for valid, red for invalid
  1154→    const alpha = 0.6;
  1155→
  1156→    this.placementMarker.lineStyle(2, color, alpha);
  1157→    this.placementMarker.beginPath();
  1158→    this.placementMarker.moveTo(topCorner.x, topCorner.y);
  1159→    this.placementMarker.lineTo(rightCorner.x, rightCorner.y);
  1160→    this.placementMarker.lineTo(bottomCorner.x, bottomCorner.y);
  1161→    this.placementMarker.lineTo(leftCorner.x, leftCorner.y);
  1162→    this.placementMarker.closePath();
  1163→    this.placementMarker.strokePath();
  1164→
  1165→    // Fill with semi-transparent color
  1166→    this.placementMarker.fillStyle(color, 0.12);
  1167→    this.placementMarker.fillPath();
  1168→
  1169→    // EXPERIMENT-OPUS-B1B2-01 / B1: Crosshair at tile center (tx, ty)
  1170→    const center = projectGroundPoint(tx, ty, this._offset);
  1171→    this.placementMarker.lineStyle(1, color, alpha * 0.7);
  1172→    this.placementMarker.beginPath();
  1173→    this.placementMarker.moveTo(center.x - 6, center.y);
  1174→    this.placementMarker.lineTo(center.x + 6, center.y);
  1175→    this.placementMarker.moveTo(center.x, center.y - 4);
  1176→    this.placementMarker.lineTo(center.x, center.y + 4);
  1177→    this.placementMarker.strokePath();
  1178→  }

  1181→   * Get the continuous fire aim target for a vehicle.
  1182→   *
  1183→   * ARENA-03H+ fixup: In Arena mode, returns null when there is no valid target.
  1184→   * This prevents continuous fire from falling back to turret-angle direction
  1185→   * or mouse position when the target is missing/destroyed.
  1186→   *
  1187→   * In non-Arena devtools mode, always returns mouse/fallback position (never null).
  1188→   *
  1189→   * @returns Aim target coordinates, or null if Arena mode has no valid target
  1190→   */
  1191→  private getContinuousFireAimTarget(vehicle: BlockoutVehicleState): { x: number; y: number } | null {
  1192→    // For non-Arena devtools mode, fall back to mouse position (never null)
  1193→    if (!this.arenaMode) {
  1194→      return {
  1195→        x: this.blockoutVehicleInputController?.mouseWorldX ?? vehicle.worldX + this._offset.x,
  1196→        y: this.blockoutVehicleInputController?.mouseWorldY ?? vehicle.worldY + this._offset.y,
  1197→      };
  1198→    }
  1199→
  1200→    // Arena mode: use target-lock
  1201→    if (vehicle.targetVehicleId) {
  1202→      const vehicles = this.gameState.blockoutVehicles;
  1203→      const target = vehicles?.find(v => v.id === vehicle.targetVehicleId);
  1204→      if (target && !target.isDestroyed) {
  1205→        const targetCenter = computeBodyWorldCenter(target, this._offset as IsoPoint);
  1206→        return { x: targetCenter.x, y: targetCenter.y };
  1207→      }
  1208→      // Target gone/destroyed — clear it and return null (stop fire)
  1209→      vehicle.targetVehicleId = null;
  1210→      return null;
  1211→    }
  1212→
  1213→    // Arena mode with no target: do not fire, do not fall back to turret angle or mouse
  1214→    return null;
  1215→  }

  1217→  /** Update Arena turret aim/rest for allies and enemies using one shared pass. */
  1218→  private updateArenaTurretAiming(delta: number): void {
  1219→    if (!this.arenaMode) return;
  1220→    const vehicles = this.gameState.blockoutVehicles;
  1221→    if (!vehicles) return;
  1222→
  1223→    for (const vehicle of vehicles) {
  1224→      if (vehicle.isDestroyed) continue;
  1225→
  1226→      let desiredAngle = vehicle.bodyAngle;
  1227→
  1228→      if (vehicle.targetVehicleId) {
  1229→        const target = vehicles.find(v => v.id === vehicle.targetVehicleId);
  1230→        if (target && !target.isDestroyed) {
  1231→          const turretMount = computeProjectedTurretMountScreen(vehicle, this._offset as IsoPoint);
  1232→          const targetCenter = computeBodyWorldCenter(target, this._offset as IsoPoint);
  1233→          desiredAngle = angleFromTo(turretMount.x, turretMount.y, targetCenter.x, targetCenter.y);
  1234→        } else {
  1235→          clearTargetAndWeaponState(vehicle);
  1236→          desiredAngle = vehicle.bodyAngle;
  1237→        }
  1238→      }
  1239→
  1240→      rotateTurretToward(vehicle, desiredAngle, delta);
  1241→    }
  1242→  }
