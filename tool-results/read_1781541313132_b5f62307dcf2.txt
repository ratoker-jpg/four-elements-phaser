     1→/**
     2→ * BlockoutVehicleRenderer — renders blockout vehicles as pseudo-isometric
     3→ * Phaser Graphics primitives following the camera projection contract.
     4→ *
     5→ * PROJECTION-01: Ground-plane retrofit.
     6→ * - Bodies rendered as pseudo-isometric boxes (base + side + top face)
     7→ * - Selection/hover rings are projected ground-plane ellipses
     8→ * - Move target markers use projected ground-plane diamonds

    11→ * - Turret sits visually on top plane using basisZ
    12→ *
    13→ * Uses no PNG assets, no asset manifest, no texture loading.
    14→ * Only active when devtools/arena mode is on.
    15→ */

   346→  // ─── Frame sync ──────────────────────────────────────────────────
   347→
   348→  /**
   349→   * Sync blockout vehicle rendering from the current blockout vehicle state.
   350→   * Called each frame. Destroys stale graphics, creates new ones for new vehicles.
   351→   */
   352→  syncFromState(vehicles: BlockoutVehicleState[]): void {
   353→    const activeIds = new Set<string>();
   354→
   355→    for (const vehicle of vehicles) {
   356→      activeIds.add(vehicle.id);
   357→
   358→      let g = this.vehicleGraphics.get(vehicle.id);
   359→      if (!g) {
   360→        g = this.scene.add.graphics();
   361→        g.setDepth(BLOCKOUT_DEPTH);
   362→        this.vehicleGraphics.set(vehicle.id, g);
   363→      }
   364→
   365→      // ── MODULAR-RUNTIME-03A: Try live modular adapter first ──────────
   366→      // When the feature flag is on and the adapter succeeds, we hide legacy
   367→      // hull/turret sprites and let the modular adapter handle positioning.
   368→      // Overlays (shadow, HP, selection, labels, weapon bars) are still drawn
   369→      // in renderVehicle() — outside the modular guard.
   370→      const modularResult = this.modularAdapter.syncVehicle(vehicle);
   371→      const useModularBody = modularResult.usedModular;
   372→
   373→      // Check for generated hull sprite (legacy path — skipped when modular active)
   374→      // PIM-HULL-WASP-DIR-MAP-01: When calibration is active and vehicle is Wasp,
   375→      // use the forced visual dir16 instead of the normal pipeline.
   376→      // Arena/devtools-only: forced direction is gated to devtools mode.
   377→      const isWaspCalibrating = !useModularBody
   378→        && this.isDevtoolsActive()
   379→        && isCalibrationActive()
   380→        && bodyIdToGeneratedHullId(vehicle.bodyId) === 'wasp'
   381→        && isOverrideActive();
   382→
   383→      let hullKey: string | null;
   384→      if (isWaspCalibrating) {
   385→        const forcedDir16 = getForcedVisualDir16() as GeneratedHullDir16Index;
   386→        hullKey = resolveGeneratedHullKeyForced(
   387→          this.scene, vehicle.bodyId, vehicle.faction,
   388→          vehicle.modificationLevel, forcedDir16,
   389→        );
   390→      } else if (useModularBody) {
   391→        hullKey = null; // modular handles its own hull
   392→      } else {
   393→        hullKey = resolveGeneratedHullKey(
   394→          this.scene, vehicle.bodyId, vehicle.faction,
   395→          vehicle.modificationLevel, vehicle.bodyAngle,
   396→        );
   397→      }
   398→      const useGeneratedHull = hullKey !== null;
   399→
   400→      // Manage hull sprite lifecycle (hide when modular takes over)
   401→      let hullSprite = this.vehicleHullSprites.get(vehicle.id);
   402→      if (useModularBody) {
   403→        // Modular active: destroy legacy hull sprite if it exists
   404→        if (hullSprite) {
   405→          hullSprite.destroy();
   406→          this.vehicleHullSprites.delete(vehicle.id);
   407→        }
   408→      } else if (useGeneratedHull && hullKey) {
   409→        if (!hullSprite) {
   410→          hullSprite = this.scene.add.image(0, 0, hullKey);
   411→          hullSprite.setScale(GENERATED_HULL_SCALE);
   412→          hullSprite.setOrigin(GENERATED_HULL_ORIGIN_X, GENERATED_HULL_ORIGIN_Y);
   413→          hullSprite.setDepth(BLOCKOUT_DEPTH + HULL_SPRITE_DEPTH_BIAS);
   414→          this.vehicleHullSprites.set(vehicle.id, hullSprite);
   415→          if (!this.generatedHullLogged) {
   416→            console.log(`[BlockoutVehicleRenderer] Using generated hull sprite for ${vehicle.bodyId}+${vehicle.weaponId}`);
   417→            this.generatedHullLogged = true;
   418→          }
   419→        } else {
   420→          hullSprite.setTexture(hullKey);
   421→        }
   422→      } else {
   423→        if (hullSprite) {
   424→          hullSprite.destroy();
   425→          this.vehicleHullSprites.delete(vehicle.id);
   426→        }
   427→        if (!this.blockoutFallbackLogged) {
   428→          console.log(`[BlockoutVehicleRenderer] No generated hull for ${vehicle.bodyId} — blockout fallback`);
   429→          this.blockoutFallbackLogged = true;
   430→        }
   431→      }
   432→
   433→      // ── RUNTIME-03: Check for generated turret sprite ──────────
   434→      // RUNTIME-03B: Gated by ENABLE_PILOT_GENERATED_TURRET_COMPOSITION.
   435→      // When the flag is false, the generated turret composition is resolved
   436→      // (preserving the pure function for audit) but no sprite is created.
   437→      // Any existing turret sprite is destroyed so the Arena returns to
   438→      // procedural fallback.
   439→      // MODULAR-RUNTIME-03A: When modular is active, skip legacy turret entirely.
   440→      const turretComp = useModularBody
   441→        ? { hasGeneratedTurret: false, turretKey: null as string | null, scale: 1, originX: 0.5, originY: 0.5, turretOffsetPx: null as { x: number; y: number } | null, socketZHeight: null as number | null }
   442→        : resolvePilotTurretComposition(
   443→            vehicle.weaponId,
   444→            vehicle.bodyId,
   445→            vehicle.faction,
   446→            vehicle.modificationLevel,
   447→            vehicle.turretAngle,
   448→            (key: string) => this.scene.textures.exists(key),
   449→          );
   450→
   451→      // Manage turret sprite lifecycle — gated by quarantine flag or modular active
   452→      let turretSprite = this.vehicleTurretSprites.get(vehicle.id);
   453→      if (useModularBody) {
   454→        // Modular active: destroy legacy turret sprite if it exists
   455→        if (turretSprite) {
   456→          turretSprite.destroy();
   457→          this.vehicleTurretSprites.delete(vehicle.id);
   458→        }
   459→        this.vehicleHasGeneratedTurret.set(vehicle.id, false);
   460→      } else if (ENABLE_PILOT_GENERATED_TURRET_COMPOSITION && turretComp.hasGeneratedTurret && turretComp.turretKey) {
   461→        if (!turretSprite) {
   462→          turretSprite = this.scene.add.image(0, 0, turretComp.turretKey);
   463→          turretSprite.setScale(turretComp.scale);
   464→          turretSprite.setOrigin(turretComp.originX, turretComp.originY);
   465→          turretSprite.setDepth(BLOCKOUT_DEPTH + HULL_SPRITE_DEPTH_BIAS + 0.1); // above hull
   466→          this.vehicleTurretSprites.set(vehicle.id, turretSprite);
   467→          this.vehicleHasGeneratedTurret.set(vehicle.id, true);
   468→          if (!this.generatedTurretLogged) {
   469→            console.log(`[BlockoutVehicleRenderer] Using generated turret sprite for ${vehicle.bodyId}+${vehicle.weaponId}`);
   470→            this.generatedTurretLogged = true;
   471→          }
   472→        } else {
   473→          turretSprite.setTexture(turretComp.turretKey);
   474→        }
   475→      } else {
   476→        // No generated turret (or quarantine active): destroy existing sprite if any
   477→        if (turretSprite) {
   478→          turretSprite.destroy();
   479→          this.vehicleTurretSprites.delete(vehicle.id);
   480→        }
   481→        this.vehicleHasGeneratedTurret.set(vehicle.id, false);
   482→      }
   483→
   484→      // RUNTIME-03A: Store turret composition for positioning in renderVehicle().
   485→      // Turret positioning is deferred until after hullSprite.setPosition() so
   486→      // we can use the actual hull sprite position as the base, and apply
   487→      // socket.zHeight through projectWorldPoint for correct elevation.
   488→      // RUNTIME-03B: Composition is still stored even when quarantine is active,
   489→      // so the data is available for audit/diagnostic purposes. The renderer
   490→      // will simply not position a turret sprite when the flag is false.
   491→      // MODULAR-RUNTIME-03A: When modular active, turretComp is a stub.
   492→      this.vehicleTurretComp.set(vehicle.id, turretComp as PilotTurretCompositionResult);
   493→
   494→      // Determine selection/hover state for this vehicle
   495→      const isSelected = vehicle.id === this._selectedVehicleId;
   496→      const isHovered = vehicle.id === this._hoveredVehicleId;
   497→
   498→      // Redraw this vehicle
   499→      this.renderVehicle(g, vehicle, isSelected, isHovered, useModularBody);
   500→
   501→      // Debug label
   502→      let label = this.debugLabels.get(vehicle.id);
   503→      if (!label && this.showDebugLabels) {
   504→        label = this.scene.add.text(0, 0, '', {
   505→          fontSize: '9px',
   506→          color: DEBUG_LABEL_COLOR,
   507→          backgroundColor: '#00000088',
   508→          padding: { x: 2, y: 1 },
   509→        });
   510→        label.setDepth(BLOCKOUT_DEPTH + 1);
   511→        label.setOrigin(0.5, 1);
   512→        this.debugLabels.set(vehicle.id, label);
   513→      }
   514→
   515→      if (label) {
   516→        const bodyCenter = computeBodyWorldCenter(vehicle, this.offset);
   517→
   518→        const selectedMarker = isSelected ? ' [SEL]' : '';
   519→        const speedMarker = vehicle.speed > 1 ? ` v=${Math.round(vehicle.speed)}` : '';
   520→        const hpMarker = vehicle.isDestroyed ? ' [DEAD]' : ` hp=${vehicle.hp}/${vehicle.maxHp}`;
   521→        label.setText(`${vehicle.bodyId}+${vehicle.weaponId}${selectedMarker}${hpMarker}${speedMarker}`);
   522→
   523→        // Position label above top face (body center + Z offset)
   524→        const labelZ = vehicle.isDestroyed ? 0 : BLOCKOUT_VEHICLE_BODY_Z + BLOCKOUT_TURRET_Z_OFFSET + 0.1;
   525→        const tilePos = unprojectScreenToGround(bodyCenter.x, bodyCenter.y, this.offset);
   526→        const labelPos = projectWorldPoint(tilePos.x, tilePos.y, labelZ, this.offset);
   527→        label.setPosition(labelPos.x, labelPos.y);
   528→        label.setVisible(this.showDebugLabels);
   529→      }
   530→    }
   531→
   532→    // ── CORE-STEP-06H+: Depth sorting ─────────────────────────────
   533→    // Sort all vehicles by isometric depth for correct occlusion.
   534→    // Vehicles with lower screen Y (further from camera) are drawn first.
   535→    const depthSortables: DepthSortable[] = vehicles
   536→      .filter(v => !v.isDestroyed && activeIds.has(v.id))
   537→      .map(v => ({
   538→        id: v.id,
   539→        type: 'unit' as const,
   540→        tx: v.tx,
   541→        ty: v.ty,
   542→        offsetX: this.offset.x,
   543→        offsetY: this.offset.y,
   544→      }));
   545→    const depthOrder = new Map<string, number>();
   546→    if (depthSortables.length > 0) {
   547→      const sorted = sortByDepth(depthSortables);
   548→      for (let i = 0; i < sorted.length; i++) {
   549→        depthOrder.set(sorted[i].sortable.id, i);
   550→      }
   551→    }
   552→
   553→    // Apply depth to each vehicle's graphics object
   554→    for (const vehicle of vehicles) {
   555→      const g = this.vehicleGraphics.get(vehicle.id);
   556→      if (g) {
   557→        const orderIdx = depthOrder.get(vehicle.id);
   558→        if (orderIdx !== undefined) {
   559→          // Use BLOCKOUT_DEPTH as base + sorted order for isometric correctness
   560→          g.setDepth(BLOCKOUT_DEPTH + orderIdx);
   561→        }
   562→      }
   563→      // Also update debug label depth
   564→      const label = this.debugLabels.get(vehicle.id);
   565→      if (label) {
   566→        const orderIdx = depthOrder.get(vehicle.id);
   567→        if (orderIdx !== undefined) {
   568→          label.setDepth(BLOCKOUT_DEPTH + orderIdx + 1);
   569→        }
   570→      }
   571→      // Also update hull sprite depth
   572→      const hullSprite = this.vehicleHullSprites.get(vehicle.id);
   573→      if (hullSprite) {
   574→        const orderIdx = depthOrder.get(vehicle.id);
   575→        if (orderIdx !== undefined) {
   576→          hullSprite.setDepth(BLOCKOUT_DEPTH + orderIdx + HULL_SPRITE_DEPTH_BIAS);
   577→        }
   578→      }
   579→      // RUNTIME-03: Also update turret sprite depth
   580→      const turretSpriteForDepth = this.vehicleTurretSprites.get(vehicle.id);
   581→      if (turretSpriteForDepth) {
   582→        const orderIdx = depthOrder.get(vehicle.id);
   583→        if (orderIdx !== undefined) {
   584→          turretSpriteForDepth.setDepth(BLOCKOUT_DEPTH + orderIdx + HULL_SPRITE_DEPTH_BIAS + 0.1);
   585→        }
   586→      }
   587→      // Also update direction debug label depth
   588→      const dirLabel = this.directionDebugLabels.get(vehicle.id);
   589→      if (dirLabel) {
   590→        const orderIdx = depthOrder.get(vehicle.id);
   591→        if (orderIdx !== undefined) {
   592→          dirLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 10);
   593→        }
   594→      }
   595→      // Also update calibration label depth
   596→      const calibLabel = this.calibrationLabels.get(vehicle.id);
   597→      if (calibLabel) {
   598→        const orderIdx = depthOrder.get(vehicle.id);
   599→        if (orderIdx !== undefined) {
   600→          calibLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 20);
   601→        }
   602→      }
   603→      // Also update placement label depth
   604→      const placeLabel = this.placementLabels.get(vehicle.id);
   605→      if (placeLabel) {
   606→        const orderIdx = depthOrder.get(vehicle.id);
   607→        if (orderIdx !== undefined) {
   608→          placeLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 25);
   609→        }
   610→      }
   611→      // MODULAR-RUNTIME-03A: Post-sort modular sprite depth resync
   612→      const modularDepthIdx = depthOrder.get(vehicle.id);
   613→      if (modularDepthIdx !== undefined) {
   614→        this.modularAdapter.setDepth(vehicle.id, modularDepthIdx);
   615→      }
   616→    }
   617→
   618→    // Clean up stale vehicles
   619→    for (const [id, g] of this.vehicleGraphics) {
   620→      if (!activeIds.has(id)) {
   621→        g.destroy();
   622→        this.vehicleGraphics.delete(id);
   623→      }
   624→    }
   625→    for (const [id, label] of this.debugLabels) {
   626→      if (!activeIds.has(id)) {
   627→        label.destroy();
   628→        this.debugLabels.delete(id);
   629→      }
   630→    }
   631→    // Clean up stale hull sprites
   632→    for (const [id, sprite] of this.vehicleHullSprites) {
   633→      if (!activeIds.has(id)) {
   634→        sprite.destroy();
   635→        this.vehicleHullSprites.delete(id);
   636→      }
   637→    }
   638→    // RUNTIME-03: Clean up stale turret sprites
   639→    for (const [id, sprite] of this.vehicleTurretSprites) {
   640→      if (!activeIds.has(id)) {
   641→        sprite.destroy();
   642→        this.vehicleTurretSprites.delete(id);
   643→      }
   644→    }
   645→    // RUNTIME-03: Clean up stale turret flags
   646→    for (const [id] of this.vehicleHasGeneratedTurret) {
   647→      if (!activeIds.has(id)) {
   648→        this.vehicleHasGeneratedTurret.delete(id);
   649→      }
   650→    }
   651→    // RUNTIME-03A: Clean up stale turret composition data
   652→    for (const [id] of this.vehicleTurretComp) {
   653→      if (!activeIds.has(id)) {
   654→        this.vehicleTurretComp.delete(id);
   655→      }
   656→    }
   657→    // MODULAR-RUNTIME-03A: Clean up stale modular sprites
   658→    // Uses removeStale() which checks the adapter's own vehicleModularSprites map,
   659→    // not the legacy vehicleHullSprites/vehicleTurretSprites (which are empty in modular mode).
   660→    this.modularAdapter.removeStale(activeIds);
   661→    // Clean up stale direction debug labels
   662→    for (const [id, label] of this.directionDebugLabels) {
   663→      if (!activeIds.has(id)) {
   664→        label.destroy();
   665→        this.directionDebugLabels.delete(id);
   666→      }
   667→    }
   668→    // Clean up stale calibration labels
   669→    for (const [id, label] of this.calibrationLabels) {
   670→      if (!activeIds.has(id)) {
   671→        label.destroy();
   672→        this.calibrationLabels.delete(id);
   673→      }
   674→    }
   675→    // Clean up stale placement labels
   676→    for (const [id, label] of this.placementLabels) {
   677→      if (!activeIds.has(id)) {
   678→        label.destroy();
   679→        this.placementLabels.delete(id);
   680→      }
   681→    }
   682→
   683→    // ── PIM-HULL-WASP-ANCHOR-MAP-01 fixup v3: On-screen calibration button panel ──
   684→    // Show the panel when placement calibration is active for a Wasp, hide otherwise.
   685→    // The panel is a fixed-screen UI that doesn't move with vehicles.
   686→    const shouldShowPanel = this.isDevtoolsActive() && isWaspPlacementActive();
   687→    if (shouldShowPanel && !this.placementPanel) {
   688→      this.placementPanel = new WaspPlacementCalibrationPanel(this.scene);
   689→      this.placementPanel.show();
   690→    } else if (!shouldShowPanel && this.placementPanel) {
   691→      this.placementPanel.hide();
   692→      this.placementPanel = null;
   693→    }
   694→  }
   695→
   696→  // ─── Vehicle rendering ──────────────────────────────────────────
   697→
   698→  private renderVehicle(g: Phaser.GameObjects.Graphics, vehicle: BlockoutVehicleState, isSelected: boolean, isHovered: boolean, useModularBody: boolean = false): void {
   699→    g.clear();
   700→
   701→    // If using modular or generated hull sprite, skip blockout body rendering but keep overlays
   702→    const hullSprite = this.vehicleHullSprites.get(vehicle.id);
   703→    const skipBlockoutBody = useModularBody || hullSprite !== undefined;
   704→
   705→    // Position hull sprite at vehicle center
   706→    if (hullSprite) {
   707→      const recoilBodyOffset = vehicle.recoilBodyOffset ?? 0;
   708→      const bodyAngle = vehicle.bodyAngle;
   709→      const bodyImpulseX = -Math.cos(bodyAngle) * recoilBodyOffset;
   710→      const bodyImpulseY = -Math.sin(bodyAngle) * recoilBodyOffset;
   711→      let spriteCx = vehicle.worldX + this.offset.x + bodyImpulseX;
   712→      let spriteCy = vehicle.worldY + this.offset.y + bodyImpulseY;
   713→
   714→      // PIM-HULL-WASP-ANCHOR-MAP-01: Apply debug placement offset for Wasp
   715→      // Arena/devtools-only: this offset is purely visual, affects sprite position only.
   716→      // Selection ring, movement, pathfinding are all unchanged.
   717→      const isWaspPlacement = this.isDevtoolsActive()
   718→        && isWaspPlacementActive()
   719→        && bodyIdToGeneratedHullId(vehicle.bodyId) === 'wasp';
   720→      if (isWaspPlacement) {
   721→        spriteCx += getWaspDebugOffsetX();
   722→        spriteCy += getWaspDebugOffsetY();
   723→      }
   724→
   725→      // PIM-WASP-SCALE-PLACEMENT-01: Apply per-hull placement offset.
   726→      // This is a permanent runtime offset (not the debug calibration offset)
   727→      // that centers the hull sprite correctly within the tile footprint
   728→      // after the scale reduction from 0.24 to 0.12.
   729→      // Only applies to hulls with generated sprites.
   730→      const hullId = bodyIdToGeneratedHullId(vehicle.bodyId);
   731→      if (hullId) {
   732→        const placement = getGeneratedHullPlacementOffset(hullId);
   733→        spriteCx += placement.offsetX;
   734→        spriteCy += placement.offsetY;
   735→      }
   736→
   737→      hullSprite.setPosition(spriteCx, spriteCy);
   738→      hullSprite.setDepth(BLOCKOUT_DEPTH + HULL_SPRITE_DEPTH_BIAS); // will be updated by depth sorting below
   739→    }
   740→
   741→    // ── RUNTIME-03A: Position turret sprite using actual hull sprite position ──
   742→    // The turret must be positioned AFTER the hull sprite has been placed so we
   743→    // use the actual hullSprite.x/y as the base (not a recomputed value that
   744→    // could diverge due to debug offsets, calibration, etc.).
   745→    // Socket zHeight is applied through projectWorldPoint for correct elevation.
   746→    const turretSprite = this.vehicleTurretSprites.get(vehicle.id);
   747→    const turretComp = this.vehicleTurretComp.get(vehicle.id);
   748→    if (turretSprite && turretComp?.turretOffsetPx && hullSprite) {
   749→      // Base position = actual hull sprite position (includes all offsets already applied)
   750→      const hullSpriteX = hullSprite.x;
   751→      const hullSpriteY = hullSprite.y;
   752→
   753→      // Apply socket zHeight through the existing projection helpers.
   754→      // The hull sprite is at ground-plane screen position. The turret mount
   755→      // point is elevated by socket.zHeight above the ground. We compute the
   756→      // z-height screen delta by projecting the hull's tile position at z=0
   757→      // and at z=socketZHeight, then taking the Y difference.
   758→      let zHeightScreenDeltaY = 0;
   759→      if (turretComp.socketZHeight !== null && turretComp.socketZHeight > 0) {
   760→        const hullTilePos = unprojectScreenToGround(hullSpriteX, hullSpriteY, this.offset);
   761→        const groundProj = projectWorldPoint(hullTilePos.x, hullTilePos.y, 0, this.offset);
   762→        const elevatedProj = projectWorldPoint(hullTilePos.x, hullTilePos.y, turretComp.socketZHeight, this.offset);
   763→        zHeightScreenDeltaY = elevatedProj.y - groundProj.y;
   764→      }
   765→
   766→      // Turret position = hull sprite position + pivot-on-socket offset + z-height elevation
   767→      const turretCx = hullSpriteX + turretComp.turretOffsetPx.x;
   768→      const turretCy = hullSpriteY + turretComp.turretOffsetPx.y + zHeightScreenDeltaY;
   769→      turretSprite.setPosition(turretCx, turretCy);
   770→    }
   771→
   772→    // ── Shared projected geometry (single source of truth) ────────
   773→    const geom = computeProjectedBlockoutVehicleGeometry(vehicle, this.offset);
   774→    const { halfW, halfH, mountTileOffset, turretHalfW, turretHalfH,
   775→            effectiveBarrelLength, effectiveTurretAngle, barrelZ,
   776→            barrelTipScreen, barrelStartScreen } = geom;
   777→
   778→    const bodyAngle = vehicle.bodyAngle;
   779→
   780→    // BLOCKOUT-04H+: Use continuous worldX/worldY + offset for position
   781→    // BLOCKOUT-05H+: Include recoil body impulse offset (shifts body backward)
   782→    const recoilBodyOffset = vehicle.recoilBodyOffset ?? 0;
   783→    const bodyImpulseX = -Math.cos(bodyAngle) * recoilBodyOffset;
   784→    const bodyImpulseY = -Math.sin(bodyAngle) * recoilBodyOffset;
   785→    const cx = vehicle.worldX + this.offset.x + bodyImpulseX;
   786→    const cy = vehicle.worldY + this.offset.y + bodyImpulseY;
   787→
   788→    // Tile-space body center (with impulse for visual rendering)
   789→    const tilePos = unprojectScreenToGround(cx, cy, this.offset);
   790→    // Tile-space mount position (shared offset + impulse-shifted body center)
   791→    const mountWorldX = tilePos.x + mountTileOffset.dx;
   792→    const mountWorldY = tilePos.y + mountTileOffset.dy;
   793→
   794→    // Faction colors
   795→    const bodyColor = FACTION_BODY_COLORS[vehicle.faction] ?? FACTION_BODY_COLORS.cyan;
   796→    const turretColor = FACTION_TURRET_COLORS[vehicle.faction] ?? FACTION_TURRET_COLORS.cyan;
   797→
   798→    // Barrel width from weapon profile
   799→    const wpProfile = getWeaponProfile(vehicle.weaponId);
   800→    const barrelWidth = wpProfile ? wpProfile.blockoutBarrelWidth : 2;
   801→
   802→    // ── Movement target marker (selected vehicle only) ──────────
   803→    if (isSelected && vehicle.hasMoveTarget) {
   804→      const targetScreenX = vehicle.targetWorldX + this.offset.x;
   805→      const targetScreenY = vehicle.targetWorldY + this.offset.y;
   806→
   807→      // Thin line from vehicle to target
   808→      g.lineStyle(1, MOVE_LINE_COLOR, MOVE_LINE_ALPHA);
   809→      g.beginPath();
   810→      g.moveTo(cx, cy);
   811→      g.lineTo(targetScreenX, targetScreenY);
   812→      g.strokePath();
   813→
   814→      // Projected ground diamond at target
   815→      g.lineStyle(1.5, MOVE_TARGET_COLOR, MOVE_TARGET_ALPHA);
   816→      drawProjectedGroundDiamond(g, targetScreenX, targetScreenY, MOVE_TARGET_DIAMOND_HALF_SIZE, this.offset);
   817→
   818→      // Projected crosshair
   819→      drawProjectedCrosshair(g, targetScreenX, targetScreenY, MOVE_TARGET_CROSSHAIR_ARM, this.offset);
   820→    }
   821→
   822→    // ── Selection highlight ring (projected ground-plane) ────────
   823→    if (isSelected) {
   824→      const pulse = 0.5 + 0.5 * Math.sin((this.scene.time.now % 800) / 800 * Math.PI * 2);
   825→      const alpha = 0.6 + 0.4 * pulse;
   826→
   827→      g.lineStyle(SELECTION_RING_WIDTH, SELECTION_RING_COLOR, alpha);
   828→      drawProjectedGroundRing(g, cx, cy, SELECTION_RING_WORLD_RADIUS, this.offset, 24);
   829→
   830→      // BLOCKOUT-10H+: Direction arrow outside the ring for orientation clarity
   831→      // Use screen-space arrow from body center along body angle
   832→      // Approximate ring edge in screen space for arrow placement
   833→      const ringEdgeDist = SELECTION_RING_WORLD_RADIUS * 38; // approximate pixel distance
   834→      const arrowBaseX = cx + Math.cos(bodyAngle) * ringEdgeDist;
   835→      const arrowBaseY = cy + Math.sin(bodyAngle) * ringEdgeDist;
   836→      const arrowTipX = cx + Math.cos(bodyAngle) * (ringEdgeDist + DIRECTION_ARROW_LENGTH);
   837→      const arrowTipY = cy + Math.sin(bodyAngle) * (ringEdgeDist + DIRECTION_ARROW_LENGTH);
   838→
   839→      // Arrow shaft
   840→      g.lineStyle(2, SELECTION_RING_COLOR, alpha);
   841→      g.beginPath();
   842→      g.moveTo(arrowBaseX, arrowBaseY);
   843→      g.lineTo(arrowTipX, arrowTipY);
   844→      g.strokePath();
   845→
   846→      // Arrow head
   847→      const headAngle1 = bodyAngle + Math.PI * 0.8;
   848→      const headAngle2 = bodyAngle - Math.PI * 0.8;
   849→      g.beginPath();
   850→      g.moveTo(arrowTipX, arrowTipY);
   851→      g.lineTo(arrowTipX + Math.cos(headAngle1) * DIRECTION_ARROW_HEAD, arrowTipY + Math.sin(headAngle1) * DIRECTION_ARROW_HEAD);
   852→      g.strokePath();
   853→      g.beginPath();
   854→      g.moveTo(arrowTipX, arrowTipY);
   855→      g.lineTo(arrowTipX + Math.cos(headAngle2) * DIRECTION_ARROW_HEAD, arrowTipY + Math.sin(headAngle2) * DIRECTION_ARROW_HEAD);
   856→      g.strokePath();
   857→    }
   858→
   859→    // ── Hover marker (projected ground-plane) ────────────────────
   860→    if (isHovered && !isSelected) {
   861→      g.lineStyle(1.5, HOVER_RING_COLOR, HOVER_RING_ALPHA);
   862→      drawProjectedGroundRing(g, cx, cy, HOVER_RING_WORLD_RADIUS, this.offset, 20);
   863→    }
   864→
   865→    // ── ARENA-03H+: Target indicator (enemy being targeted by selected ally) ──
   866→    const isTargeted = vehicle.id === this._targetedVehicleId;
   867→    if (isTargeted && !isSelected) {
   868→      const pulse = 0.5 + 0.5 * Math.sin((this.scene.time.now % 600) / 600 * Math.PI * 2);
   869→      const alpha = 0.5 + 0.5 * pulse;
   870→      g.lineStyle(2, 0xff4444, alpha); // Red targeting ring
   871→      drawProjectedGroundRing(g, cx, cy, SELECTION_RING_WORLD_RADIUS, this.offset, 24);
   872→
   873→      // Small crosshair in center
   874→      const crossSize = 0.12;
   875→      g.lineStyle(1.5, 0xff4444, alpha);
   876→      drawProjectedCrosshair(g, cx, cy, crossSize, this.offset);
   877→    }
   878→
   879→    // ── CORE-STEP-07H+: Target-lock status indicator on attacker ──
   880→    // Show a small colored dot above the turret when this vehicle has an active target-lock
   881→    if (vehicle.targetVehicleId && !vehicle.isDestroyed) {
   882→      // Target-lock active: show yellow dot above turret
   883→      const lockIndicatorZ = BLOCKOUT_VEHICLE_BODY_Z + BLOCKOUT_TURRET_Z_OFFSET + 0.3;
   884→      const tilePosLocal = unprojectScreenToGround(cx, cy, this.offset);
   885→      const lockPos = projectWorldPoint(tilePosLocal.x, tilePosLocal.y, lockIndicatorZ, this.offset);
   886→      g.fillStyle(0xffcc00, 0.9); // Yellow target-lock indicator
   887→      g.fillCircle(lockPos.x, lockPos.y, 3);
   888→    }
   889→
   890→    // ── ARENA-03H+: Enemy team indicator (small red diamond above HP bar) ──
   891→    if (vehicle.team === 'enemy' && !vehicle.isDestroyed) {
   892→      const indicatorZ = BLOCKOUT_VEHICLE_BODY_Z + BLOCKOUT_TURRET_Z_OFFSET + 0.2;
   893→      const indicatorPos = projectWorldPoint(tilePos.x, tilePos.y, indicatorZ, this.offset);
   894→      const indicatorSize = 0.08;
   895→      g.lineStyle(1, 0xff4444, 0.7);
   896→      drawProjectedGroundDiamond(g, indicatorPos.x, indicatorPos.y, indicatorSize, this.offset);
   897→    }
   898→
   899→    // ── BLOCKOUT-07H+: Destroyed vehicle rendering ────────────────
   900→    if (vehicle.isDestroyed) {
   901→      // Dimmed flat body on ground (no height)
   902→      const cosA = Math.cos(bodyAngle);
   903→      const sinA = Math.sin(bodyAngle);
   904→      const localCorners = [
   905→        { lx: -halfW, ly: -halfH },
   906→        { lx: halfW, ly: -halfH },
   907→        { lx: halfW, ly: halfH },
   908→        { lx: -halfW, ly: halfH },
   909→      ];
   910→      const basePts = localCorners.map(c => {
   911→        const wx = tilePos.x + c.lx * cosA - c.ly * sinA;
   912→        const wy = tilePos.y + c.lx * sinA + c.ly * cosA;
   913→        return projectWorldPoint(wx, wy, 0, this.offset);
   914→      });
   915→
   916→      g.fillStyle(bodyColor, 0.3);
   917→      g.beginPath();
   918→      g.moveTo(basePts[0].x, basePts[0].y);
   919→      for (let i = 1; i < basePts.length; i++) {
   920→        g.lineTo(basePts[i].x, basePts[i].y);
   921→      }
   922→      g.closePath();
   923→      g.fillPath();
   924→
   925→      g.lineStyle(1, BODY_OUTLINE_COLOR, 0.5);
   926→      g.beginPath();
   927→      g.moveTo(basePts[0].x, basePts[0].y);
   928→      for (let i = 1; i < basePts.length; i++) {
   929→        g.lineTo(basePts[i].x, basePts[i].y);
   930→      }
   931→      g.closePath();
   932→      g.strokePath();
   933→
   934→      // X marker over body (use tile-unit size for consistency)
   935→      g.lineStyle(2, 0xff0000, 0.8);
   936→      const xSize = Math.min(halfW, halfH) * PROJ_TILE_W / 2 - 2;
   937→      g.beginPath();
   938→      g.moveTo(cx - xSize, cy - xSize);
   939→      g.lineTo(cx + xSize, cy + xSize);
   940→      g.strokePath();
   941→      g.beginPath();
   942→      g.moveTo(cx + xSize, cy - xSize);
   943→      g.lineTo(cx - xSize, cy + xSize);
   944→      g.strokePath();
   945→
   946→      // No turret/barrel or HP bar for destroyed vehicles
   947→      return;
   948→    }
   949→
   950→    // ── Vehicle shadow (projected ground-plane) ──────────────────
   951→    // ── Pseudo-isometric body (base + side + top) ────────────────
   952→    if (!skipBlockoutBody) {
   953→      const shadowRadius = Math.max(halfW, halfH) * SHADOW_RADIUS_FRACTION;
   954→      drawProjectedShadow(g, cx, cy, shadowRadius, this.offset);
   955→
   956→      // Derive side color (darker than body) and top color (brighter)
   957→      const sideR = ((bodyColor >> 16) & 0xff) * 0.6;
   958→      const sideG = ((bodyColor >> 8) & 0xff) * 0.6;
   959→      const sideB = (bodyColor & 0xff) * 0.6;
   960→      const sideColor = (Math.floor(sideR) << 16) | (Math.floor(sideG) << 8) | Math.floor(sideB);
   961→
   962→      const topR = Math.min(255, ((bodyColor >> 16) & 0xff) * 1.2);
   963→      const topG = Math.min(255, ((bodyColor >> 8) & 0xff) * 1.2);
   964→      const topB = Math.min(255, (bodyColor & 0xff) * 1.2);
   965→      const topColor = (Math.floor(topR) << 16) | (Math.floor(topG) << 8) | Math.floor(topB);
   966→
   967→      drawProjectedBox(
   968→        g, cx, cy, halfW, halfH, BLOCKOUT_VEHICLE_BODY_Z,
   969→        this.offset, bodyAngle,
   970→        bodyColor, sideColor, topColor, BODY_OUTLINE_COLOR,
   971→        0.3, 0.75, 1.0,
   972→      );
