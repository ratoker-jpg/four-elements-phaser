   304→  // ─── Frame sync ──────────────────────────────────────────────────
   305→
   306→  /**
   307→   * Sync blockout vehicle rendering from the current blockout vehicle state.
   308→   * Called each frame. Destroys stale graphics, creates new ones for new vehicles.
   309→   */
   310→  syncFromState(vehicles: BlockoutVehicleState[]): void {
   311→    const activeIds = new Set<string>();
   312→
   313→    for (const vehicle of vehicles) {
   314→      activeIds.add(vehicle.id);
   315→
   316→      let g = this.vehicleGraphics.get(vehicle.id);
   317→      if (!g) {
   318→        g = this.scene.add.graphics();
   319→        g.setDepth(BLOCKOUT_DEPTH);
   320→        this.vehicleGraphics.set(vehicle.id, g);
   321→      }
   322→
   323→      // Check for generated hull sprite
   324→      // PIM-HULL-WASP-DIR-MAP-01: When calibration is active and vehicle is Wasp,
   325→      // use the forced visual dir16 instead of the normal pipeline.
   326→      // Arena/devtools-only: forced direction is gated to devtools mode.
   327→      const isWaspCalibrating = this.isDevtoolsActive()
   328→        && isCalibrationActive()
   329→        && bodyIdToGeneratedHullId(vehicle.bodyId) === 'wasp'
   330→        && isOverrideActive();
   331→
   332→      let hullKey: string | null;
   333→      if (isWaspCalibrating) {
   334→        const forcedDir16 = getForcedVisualDir16() as GeneratedHullDir16Index;
   335→        hullKey = resolveGeneratedHullKeyForced(
   336→          this.scene, vehicle.bodyId, vehicle.faction,
   337→          vehicle.modificationLevel, forcedDir16,
   338→        );
   339→      } else {
   340→        hullKey = resolveGeneratedHullKey(
   341→          this.scene, vehicle.bodyId, vehicle.faction,
   342→          vehicle.modificationLevel, vehicle.bodyAngle,
   343→        );
   344→      }
   345→      const useGeneratedHull = hullKey !== null;
   346→
   347→      // Manage hull sprite lifecycle
   348→      let hullSprite = this.vehicleHullSprites.get(vehicle.id);
   349→      if (useGeneratedHull && hullKey) {
   350→        if (!hullSprite) {
   351→          hullSprite = this.scene.add.image(0, 0, hullKey);
   352→          hullSprite.setScale(GENERATED_HULL_SCALE);
   353→          hullSprite.setOrigin(GENERATED_HULL_ORIGIN_X, GENERATED_HULL_ORIGIN_Y);
   354→          hullSprite.setDepth(BLOCKOUT_DEPTH + HULL_SPRITE_DEPTH_BIAS);
   355→          this.vehicleHullSprites.set(vehicle.id, hullSprite);
   356→          if (!this.generatedHullLogged) {
   357→            console.log(`[BlockoutVehicleRenderer] Using generated hull sprite for ${vehicle.bodyId}+${vehicle.weaponId}`);
   358→            this.generatedHullLogged = true;
   359→          }
   360→        } else {
   361→          hullSprite.setTexture(hullKey);
   362→        }
   363→      } else {
   364→        if (hullSprite) {
   365→          hullSprite.destroy();
   366→          this.vehicleHullSprites.delete(vehicle.id);
   367→        }
   368→        if (!this.blockoutFallbackLogged) {
   369→          console.log(`[BlockoutVehicleRenderer] No generated hull for ${vehicle.bodyId} — blockout fallback`);
   370→          this.blockoutFallbackLogged = true;
   371→        }
   372→      }
   373→
   374→      // Determine selection/hover state for this vehicle
   375→      const isSelected = vehicle.id === this._selectedVehicleId;
   376→      const isHovered = vehicle.id === this._hoveredVehicleId;
   377→
   378→      // Redraw this vehicle
   379→      this.renderVehicle(g, vehicle, isSelected, isHovered);
   380→
   381→      // Debug label
   382→      let label = this.debugLabels.get(vehicle.id);
   383→      if (!label && this.showDebugLabels) {
   384→        label = this.scene.add.text(0, 0, '', {
   385→          fontSize: '9px',
   386→          color: DEBUG_LABEL_COLOR,
   387→          backgroundColor: '#00000088',
   388→          padding: { x: 2, y: 1 },
   389→        });
   390→        label.setDepth(BLOCKOUT_DEPTH + 1);
   391→        label.setOrigin(0.5, 1);
   392→        this.debugLabels.set(vehicle.id, label);
   393→      }
   394→
   395→      if (label) {
   396→        const bodyCenter = computeBodyWorldCenter(vehicle, this.offset);
   397→
   398→        const selectedMarker = isSelected ? ' [SEL]' : '';
   399→        const speedMarker = vehicle.speed > 1 ? ` v=${Math.round(vehicle.speed)}` : '';
   400→        const hpMarker = vehicle.isDestroyed ? ' [DEAD]' : ` hp=${vehicle.hp}/${vehicle.maxHp}`;
   401→        label.setText(`${vehicle.bodyId}+${vehicle.weaponId}${selectedMarker}${hpMarker}${speedMarker}`);
   402→
   403→        // Position label above top face (body center + Z offset)
   404→        const labelZ = vehicle.isDestroyed ? 0 : BLOCKOUT_VEHICLE_BODY_Z + BLOCKOUT_TURRET_Z_OFFSET + 0.1;
   405→        const tilePos = unprojectScreenToGround(bodyCenter.x, bodyCenter.y, this.offset);
   406→        const labelPos = projectWorldPoint(tilePos.x, tilePos.y, labelZ, this.offset);
   407→        label.setPosition(labelPos.x, labelPos.y);
   408→        label.setVisible(this.showDebugLabels);
   409→      }
   410→    }
   411→
   412→    // ── CORE-STEP-06H+: Depth sorting ─────────────────────────────
   413→    // Sort all vehicles by isometric depth for correct occlusion.
   414→    // Vehicles with lower screen Y (further from camera) are drawn first.
   415→    const depthSortables: DepthSortable[] = vehicles
   416→      .filter(v => !v.isDestroyed && activeIds.has(v.id))
   417→      .map(v => ({
   418→        id: v.id,
   419→        type: 'unit' as const,
   420→        tx: v.tx,
   421→        ty: v.ty,
   422→        offsetX: this.offset.x,
   423→        offsetY: this.offset.y,
   424→      }));
   425→    const depthOrder = new Map<string, number>();
   426→    if (depthSortables.length > 0) {
   427→      const sorted = sortByDepth(depthSortables);
   428→      for (let i = 0; i < sorted.length; i++) {
   429→        depthOrder.set(sorted[i].sortable.id, i);
   430→      }
   431→    }
   432→
   433→    // Apply depth to each vehicle's graphics object
   434→    for (const vehicle of vehicles) {
   435→      const g = this.vehicleGraphics.get(vehicle.id);
   436→      if (g) {
   437→        const orderIdx = depthOrder.get(vehicle.id);
   438→        if (orderIdx !== undefined) {
   439→          // Use BLOCKOUT_DEPTH as base + sorted order for isometric correctness
   440→          g.setDepth(BLOCKOUT_DEPTH + orderIdx);
   441→        }
   442→      }
   443→      // Also update debug label depth
   444→      const label = this.debugLabels.get(vehicle.id);
   445→      if (label) {
   446→        const orderIdx = depthOrder.get(vehicle.id);
   447→        if (orderIdx !== undefined) {
   448→          label.setDepth(BLOCKOUT_DEPTH + orderIdx + 1);
   449→        }
   450→      }
   451→      // Also update hull sprite depth
   452→      const hullSprite = this.vehicleHullSprites.get(vehicle.id);
   453→      if (hullSprite) {
   454→        const orderIdx = depthOrder.get(vehicle.id);
   455→        if (orderIdx !== undefined) {
   456→          hullSprite.setDepth(BLOCKOUT_DEPTH + orderIdx + HULL_SPRITE_DEPTH_BIAS);
   457→        }
   458→      }
   459→      // Also update direction debug label depth
   460→      const dirLabel = this.directionDebugLabels.get(vehicle.id);
   461→      if (dirLabel) {
   462→        const orderIdx = depthOrder.get(vehicle.id);
   463→        if (orderIdx !== undefined) {
   464→          dirLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 10);
   465→        }
   466→      }
   467→      // Also update calibration label depth
   468→      const calibLabel = this.calibrationLabels.get(vehicle.id);
   469→      if (calibLabel) {
   470→        const orderIdx = depthOrder.get(vehicle.id);
   471→        if (orderIdx !== undefined) {
   472→          calibLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 20);
   473→        }
   474→      }
   475→      // Also update placement label depth
   476→      const placeLabel = this.placementLabels.get(vehicle.id);
   477→      if (placeLabel) {
   478→        const orderIdx = depthOrder.get(vehicle.id);
   479→        if (orderIdx !== undefined) {
   480→          placeLabel.setDepth(BLOCKOUT_DEPTH + orderIdx + 25);
   481→        }
   482→      }
   483→    }
   484→
   485→    // Clean up stale vehicles
   486→    for (const [id, g] of this.vehicleGraphics) {
   487→      if (!activeIds.has(id)) {
   488→        g.destroy();
   489→        this.vehicleGraphics.delete(id);
   490→      }
   491→    }
   492→    for (const [id, label] of this.debugLabels) {
   493→      if (!activeIds.has(id)) {
   494→        label.destroy();
   495→        this.debugLabels.delete(id);
   496→      }
   497→    }
   498→    // Clean up stale hull sprites
   499→    for (const [id, sprite] of this.vehicleHullSprites) {
   500→      if (!activeIds.has(id)) {
   501→        sprite.destroy();
   502→        this.vehicleHullSprites.delete(id);
   503→      }
   504→    }
   505→    // Clean up stale direction debug labels
   506→    for (const [id, label] of this.directionDebugLabels) {
   507→      if (!activeIds.has(id)) {
   508→        label.destroy();
   509→        this.directionDebugLabels.delete(id);
   510→      }
   511→    }
   512→    // Clean up stale calibration labels
   513→    for (const [id, label] of this.calibrationLabels) {
   514→      if (!activeIds.has(id)) {
   515→        label.destroy();
   516→        this.calibrationLabels.delete(id);
   517→      }
   518→    }
   519→    // Clean up stale placement labels
   520→    for (const [id, label] of this.placementLabels) {
   521→      if (!activeIds.has(id)) {
   522→        label.destroy();
   523→        this.placementLabels.delete(id);
   524→      }
   525→    }
   526→
   527→    // ── PIM-HULL-WASP-ANCHOR-MAP-01 fixup v3: On-screen calibration button panel ──
   528→    // Show the panel when placement calibration is active for a Wasp, hide otherwise.
   529→    // The panel is a fixed-screen UI that doesn't move with vehicles.
   530→    const shouldShowPanel = this.isDevtoolsActive() && isWaspPlacementActive();
   531→    if (shouldShowPanel && !this.placementPanel) {
   532→      this.placementPanel = new WaspPlacementCalibrationPanel(this.scene);
   533→      this.placementPanel.show();
   534→    } else if (!shouldShowPanel && this.placementPanel) {
   535→      this.placementPanel.hide();
   536→      this.placementPanel = null;
   537→    }
   538→  }

   540→  // ─── Vehicle rendering ──────────────────────────────────────────
   541→
   542→  private renderVehicle(g: Phaser.GameObjects.Graphics, vehicle: BlockoutVehicleState, isSelected: boolean, isHovered: boolean): void {
   543→    g.clear();
   544→
   545→    // If using generated hull sprite, skip blockout body rendering but keep overlays
   546→    const hullSprite = this.vehicleHullSprites.get(vehicle.id);
   547→    const skipBlockoutBody = hullSprite !== undefined;
   548→
   549→    // Position hull sprite at vehicle center
   550→    if (hullSprite) {
   551→      const recoilBodyOffset = vehicle.recoilBodyOffset ?? 0;
   552→      const bodyAngle = vehicle.bodyAngle;

  1003→      });

  1010→      // Turret side face (left: base[3]→base[0] → top[3]→top[0])

  1020→      // Turret side face (right: base[0]→base[1] → top[0]→top[1])

  1026→      g.lineTo(turretTopPts[0].x, turretTopPts[0].y);
  1027→      g.closePath();
  1028→      g.fillPath();
  1029→

  1031→      g.fillStyle(turretColor, 1);
  1032→      g.beginPath();

  1046→        g.lineTo(turretTopPts[i].x, turretTopPts[i].y);
  1047→      }

  1049→      g.strokePath();

  1053→      g.beginPath();

  1055→      g.lineTo(barrelTipScreen.x, barrelTipScreen.y);
  1056→      g.strokePath();

  1061→        const aimTileLength = AIM_LINE_LENGTH / PROJ_TILE_W;

  1064→        const aimEndWorld = {
  1065→          x: mountWorldX + (turretHalfW + effectiveBarrelLength + aimTileLength) * turretCosA,
  1066→          y: mountWorldY + (turretHalfW + effectiveBarrelLength + aimTileLength) * turretSinA,
  1067→        };
  1068→        const aimEnd = projectWorldPoint(aimEndWorld.x, aimEndWorld.y, barrelZ, this.offset);
  1069→

  1071→        const dashLen = AIM_LINE_DASH;
  1072→        const gapLen = AIM_LINE_GAP;
  1073→        const dx = aimEnd.x - aimStart.x;
  1074→        const dy = aimEnd.y - aimStart.y;
  1075→        const totalLen = Math.sqrt(dx * dx + dy * dy);
  1076→        const ux = dx / totalLen;
  1077→        const uy = dy / totalLen;

  1080→        while (pos < totalLen) {
  1081→          const segEnd = Math.min(pos + dashLen, totalLen);
  1082→          g.beginPath();
  1083→          g.moveTo(aimStart.x + ux * pos, aimStart.y + uy * pos);
  1084→          g.lineTo(aimStart.x + ux * segEnd, aimStart.y + uy * segEnd);
  1085→          g.strokePath();
  1086→          pos = segEnd + gapLen;
  1087→        }
  1088→      }
  1089→    }
